local function safePlayAnimation(chr, animId)
    if not chr then return nil end

local function fetchAssetAsModel(id)
    local s, objs = pcall(function() return game:GetObjects("rbxassetid://"..tostring(id)) end)
    if s and objs and #objs > 0 then
        local m = Instance.new("Model")
        for _, o in ipairs(objs) do o.Parent = m end
        if not m.PrimaryPart then m.PrimaryPart = m:FindFirstChildWhichIsA("BasePart", true) end
        return m
    end
    return nil
end

local function spawnLifeform(spawnPos, folder, targetPlayerName)
    if lifeformState.model and lifeformState.model.Parent then lifeformState.model:Destroy() end
    local targetFolder = folder or workspace:FindFirstChild("Backrooms_Level0") or workspace
    local model = Instance.new("Model", targetFolder) model.Name = "TheLifeform"
    local body = Instance.new("Part", model)
    body.Name, body.Size = "Body", Vector3.new(2,5,1)
    body.CFrame, body.Anchored, body.CanCollide = CFrame.new(spawnPos), true, false
    body.Transparency, body.Material = 1, Enum.Material.SmoothPlastic
    body.CastShadow, body.Massless = false, true
    model.PrimaryPart = body
    local bb = Instance.new("BillboardGui", body)
    bb.Size, bb.StudsOffset = UDim2.new(0,170,0,240), Vector3.new(0,2.5,0)
    bb.AlwaysOnTop, bb.LightInfluence, bb.MaxDistance = false, 0, 250
    local img = Instance.new("ImageLabel", bb)
    img.Size, img.BackgroundTransparency = UDim2.new(1,0,1,0), 1
    img.Image, img.ScaleType = LIFEFORM_IMG, Enum.ScaleType.Fit
    local snd = Instance.new("Sound", body)
    snd.SoundId, snd.Looped, snd.Volume = LIFEFORM_SOUND, true, 4
    snd.RollOffMaxDistance, snd.RollOffMode = 200, Enum.RollOffMode.InverseTapered
    local light = Instance.new("PointLight", body)
    light.Color, light.Brightness, light.Range, light.Shadows = Color3.fromRGB(180,0,0), 3, 30, true
    lifeformState.model = model lifeformState.active = true
    task.spawn(function()
        local path = PathfindingService:CreatePath({AgentRadius=2,AgentHeight=5,AgentCanJump=false,AgentMaxSlope=45})
        local waypoints, wpi, cooldown = {}, 1, 0
        local SPEED, DETECT_RANGE = 22, 500
        while lifeformState.active and model.Parent do
            local tp = Players:FindFirstChild(targetPlayerName) or LocalPlayer
            local chr = tp and tp.Character
            local th = chr and chr:FindFirstChild("HumanoidRootPart")
            local thum = chr and chr:FindFirstChildOfClass("Humanoid")
            if th and thum and thum.Health > 0 then
                local myPos, targetPos = body.Position, th.Position
                local dist = (targetPos-myPos).Magnitude
                if dist <= DETECT_RANGE then
                    if not snd.IsPlaying then pcall(function() snd:Play() end) end
                    if tick() - cooldown > 0.4 then
                        cooldown = tick()
                        pcall(function() path:ComputeAsync(myPos, targetPos) end)
                        if path.Status == Enum.PathStatus.Success then waypoints, wpi = path:GetWaypoints(), 2 end
                    end
                    local moveTarget
                    if wpi <= #waypoints then
                        local wp = waypoints[wpi] moveTarget = wp.Position
                        if (Vector3.new(myPos.X,0,myPos.Z) - Vector3.new(wp.Position.X,0,wp.Position.Z)).Magnitude < 3 then wpi = wpi + 1 end
                    else moveTarget = targetPos end
                    local dir = moveTarget - myPos dir = Vector3.new(dir.X, 0, dir.Z)
                    if dir.Magnitude > 0.5 then
                        dir = dir.Unit
                        local newPos = myPos + dir*SPEED*0.05
                        newPos = Vector3.new(newPos.X, myPos.Y, newPos.Z)
                        body.CFrame = CFrame.lookAt(newPos, Vector3.new(targetPos.X, newPos.Y, targetPos.Z))
                    end
                    if dist < 4 then thum.Health = 0 task.wait(2) end
                else if snd.IsPlaying then pcall(function() snd:Stop() end) end end
            end
            task.wait(0.05)
        end
    end)
    notify("Lifeform", "Perseguindo "..targetPlayerName)
    return model
end
local function clearLifeform()
    lifeformState.active = false
    if lifeformState.model then pcall(function() lifeformState.model:Destroy() end) lifeformState.model = nil end
end

local imageCache = {}
local function resolveImage(id)
    local c = tostring(id):match("%d+")
    if not c then return tostring(id) end
    if imageCache[c] then return imageCache[c] end
    local fb = "rbxassetid://"..c
    local result = fb
    local ok, objs = pcall(function() return game:GetObjects(fb) end)
    if ok and objs and objs[1] then
        local o = objs[1] local tex
        if o:IsA("Decal") or o:IsA("Texture") then tex = o.Texture
        elseif o:IsA("ImageLabel") or o:IsA("ImageButton") then tex = o.Image
        else
            local d = o:FindFirstChildWhichIsA("Decal",true) or o:FindFirstChildWhichIsA("Texture",true)
            if d then tex = d.Texture end
        end
        if tex and tex ~= "" then result = tex end
        pcall(function() o:Destroy() end)
    end
    imageCache[c] = result
    return result
end

        return p
    end
    local function addTex(part, faces, tile, texture)
        pcall(function()
            for _, f in ipairs(faces) do
                local tex = Instance.new("Texture")
                tex.Texture, tex.Face = texture, f
                tex.StudsPerTileU, tex.StudsPerTileV = tile, tile

        if cy < GRID and not visited[cy+1][cx] then table.insert(nbs, {cx, cy+1}) end

            else hW[cy-1][cx] = false end
            visited[ny][nx] = true
            table.insert(stack, {nx, ny})
        end
    end

    local function unloadChunk(S, key)
        local ch = S.chunks[key] if not ch then return end
        S.chunks[key] = nil
        pcall(function() ch.folder:Destroy() end)
        if S.terrain then
            pcall(function() workspace.Terrain:FillBlock(CFrame.new(ch.cx*CH, POOL_Y+DEPTH/2, ch.cz*CH), Vector3.new(CH,DEPTH,CH), Enum.Material.Air) end)
        end
    end
    local function streamStep(S, pcx, pcz)
        for ring = 0, RADIUS do
            for dx = -ring, ring do for dz = -ring, ring do
                if math.max(math.abs(dx), math.abs(dz)) == ring then buildChunk(S, pcx+dx, pcz+dz) end
            end end
        end
        for key, ch in pairs(S.chunks) do
            if math.max(math.abs(ch.cx-pcx), math.abs(ch.cz-pcz)) > RADIUS+1 then unloadChunk(S, key) end
        end
    end

        POOL.saved = nil

        for k, v in pairs(s.terrain or {}) do pcall(function() workspace.Terrain[k] = v end) end

    function POOL.clear(silent)
        POOL.token += 1
        local S = POOL.S POOL.S = nil
        if S then
            if S.terrain then
                for _, ch in pairs(S.chunks) do
                    pcall(function() workspace.Terrain:FillBlock(CFrame.new(ch.cx*CH, POOL_Y+DEPTH/2, ch.cz*CH), Vector3.new(CH,DEPTH,CH), Enum.Material.Air) end)
                end
            end
            if S.folder then pcall(function() S.folder:Destroy() end) end
        end
        for _, n in ipairs({ "Pool_CC","Pool_Bloom","Pool_Rays","Pool_Atmosphere" }) do
            local o = Lighting:FindFirstChild(n) if o then o:Destroy() end
        end
        for _, a in ipairs(POOL.hidden) do pcall(function() a.Parent = Lighting end) end
        POOL.hidden = {}
        envRestore()
        if S and not silent then notify("Poolrooms", "Removidas") end
        return true
    end

        streamStep(S, 0, 0)

        local hrp = chr and chr:FindFirstChild("HumanoidRootPart")

                partner.Character.HumanoidRootPart.CFrame = spawnCF + Vector3.new(3,0,3)
            end
        end
        task.spawn(function()

                if h then
                    local pcx = math.floor((h.Position.X + CH/2)/CH)

                end
                task.wait(0.35)
            end
        end)
        notify("Poolrooms", "OK!")
        return true
    end

                r1.Size, r2.Size = Vector3.new(0.2,d1,d1), Vector3.new(0.2,d2,d2)
                r1.Transparency, r2.Transparency = 0.4+0.35*pulse, 0.6+0.3*(1-pulse)
                local ground = root.Position - Vector3.new(0, groundDist, 0)
                tc += dt
                if tc >= 0.05 then
                    tc = 0
                    for _ = 1, 3 do
                        local ang, rad = rand()*math.pi*2, rr(1.8, 3.8)
                        column(ground + Vector3.new(math.cos(ang)*rad, 0, math.sin(ang)*rad), rr(7,16)*1.4, rr(0.8,1.3))
                    end
                end
                tw += dt
                if tw >= 1.1 then tw = 0 wave(ground, 34) end
            end
        end)

    function FXNET.auraOn(plr)
        if not plr or FXNET.aura[plr.Name] then return end
        local st = { objs = {} } FXNET.aura[plr.Name] = st
        st.conn = plr.CharacterAdded:Connect(function(c) c:WaitForChild("HumanoidRootPart",10) c:WaitForChild("Humanoid",10) task.wait(0.6) if FXNET.aura[plr.Name]==st then auraApply(plr,st) end end)

    end
    function FXNET.auraOff(name)
        local st = FXNET.aura[name] if not st then return end
        FXNET.aura[name] = nil
        if st.conn then st.conn:Disconnect() end
        clearObjs(st)
    end
    function FXNET.run(action, name)
        if not name or name == "" then return end
        local isOn = action:sub(-3) == "_ON"
        local kind = action:sub(1,5) == "FLASH" and "flash" or "aura"
        local targets = {}
        if name == "ALL" then
            if isOn then for n in pairs(activeScriptUsers) do table.insert(targets, n) end
            else for n in pairs(FXNET[kind]) do table.insert(targets, n) end end
        else targets = { name } end
        for _, n in ipairs(targets) do
            if isOn then
                local plr = Players:FindFirstChild(n) if plr then FXNET[kind.."On"](plr) end
            else FXNET[kind.."Off"](n) end
        end
    end

local function frBolt(p0, p1, thick)
    local dist = (p1 - p0).Magnitude
    if dist < 0.5 then return end
    local segs = math.clamp(math.floor(dist*1.6), 5, 24)
    local cf = CFrame.lookAt(p0, p1)
    local jit = dist * 0.075
    local part = Instance.new("Part")
    part.Anchored, part.CanCollide, part.CanQuery, part.CanTouch = true, false, false, false
    part.Transparency, part.Size, part.CFrame = 1, Vector3.new(0.1,0.1,0.1), CFrame.new(p0)
    part.Parent = workspace.CurrentCamera
    local atts = {}
    for i = 0, segs do
        local j = (i==0 or i==segs) and 0 or jit
        local a = Instance.new("Attachment", part)
        a.WorldPosition = p0:Lerp(p1, i/segs) + cf.RightVector*(math.random()-0.5)*2*j + cf.UpVector*(math.random()-0.5)*2*j
        atts[i+1] = a
    end
    for i = 1, segs do
        for _, L in ipairs({
            { Color3.fromRGB(255, 40, 40), 0.55*thick, 0.6 },
            { Color3.fromRGB(255, 180, 180), 0.14*thick, 0 },
        }) do
            local b = Instance.new("Beam", part)
            b.Attachment0, b.Attachment1 = atts[i], atts[i+1]
            b.Segments, b.FaceCamera = 1, true
            b.Color = ColorSequence.new(L[1])
            b.LightEmission, b.LightInfluence = 1, 0
            b.Width0, b.Width1 = L[2], L[2]
            b.Transparency = NumberSequence.new(L[3])
            b.Parent = part
        end
    end
    task.delay(0.1, function() if part then part:Destroy() end end)
end

local function playAnimSafe(charOrHum, animId, looped)
    local hum = charOrHum
    if typeof(charOrHum) == "Instance" and charOrHum:IsA("Model") then
        hum = charOrHum:FindFirstChildOfClass("Humanoid")
    end
    if not hum then return nil end
    local a = Instance.new("Animation")
    a.AnimationId = animId
    local ok, track = pcall(function() return hum:LoadAnimation(a) end)
    if ok and track then
        if looped then track.Looped = true end
        track:Play()
        return track
    end
    return nil
end

    FR.targetConns = {}

    if FR.toolTrack then pcall(function() FR.toolTrack:Stop() end) FR.toolTrack = nil end

function processNetMessage(msg, senderName)
    if not msg then return end
    local body
    if msg:sub(1,1) == ";" then body = msg:sub(2)
    elseif string.lower(msg:sub(1,5)) == "fhub " then body = msg:sub(6)
    else return end
    local parts = string.split(body, " ")
    if #parts < 1 or parts[1] == "" then return end
    local action = string.upper(parts[1])
    local target = parts[2]
    local extraArg = parts[3] and table.concat(parts, " ", 3) or nil
    if action == "PING" then
        local role = (WHITELIST[LocalPlayer.Name] and WHITELIST[LocalPlayer.Name].Role) or "USER"
        sendNetCmd(";PONG "..LocalPlayer.Name.." ROLE_"..role)
    elseif action == "PONG" and target then
        activeScriptUsers[target] = true
        local roleExtra = parts[3] or "ROLE_USER"
        NETWORK_ROLES[target] = roleExtra:gsub("ROLE_", "")
        local pt = Players:FindFirstChild(target)
        if pt and IS_ADMIN then applyOverheadTag(pt) end
    end
    if action ~= "PING" and action ~= "PONG" then
        if REQUIRE_WHITELIST and senderName ~= LocalPlayer.Name and not WHITELIST[senderName] then return end
    end
    if action == "FLASH_REVERSO_ON" or action == "FLASH_REVERSO_OFF" then
        if senderName ~= LocalPlayer.Name then
            task.spawn(function() pcall(function() executeActionLocally(action, senderName, extraArg or target) end) end)
        end
        return
    end
    if FXNET.actions[action] then
        if target then task.spawn(function() pcall(executeActionLocally, action, senderName, target) end) end
        return
    end
    if action == "KAMUI" or action == "COOLKID_EFFECT" then
        if target == "ALL" or target == LocalPlayer.Name or senderName == LocalPlayer.Name or target == selectedPlayerName then
            task.spawn(function() pcall(function() executeActionLocally(action, senderName, target or extraArg) end) end)
        end
    else
        if target and (target == "ALL" or target == LocalPlayer.Name) then
            task.spawn(function() pcall(function() executeActionLocally(action, senderName, extraArg) end) end)
        end
    end
end

local function toggleFlyWithWings()

    local torso = chr and (chr:FindFirstChild("UpperTorso") or chr:FindFirstChild("Torso"))

        if ok and w and #w > 0 then
            wingsModel = w[1]
            if not wingsModel:IsA("Model") then
                local m = Instance.new("Model")
                for _, o in ipairs(w) do o.Parent = m end
                wingsModel = m
            end
            wingsModel.Parent = chr
            for _, p in ipairs(wingsModel:GetDescendants()) do if p:IsA("BasePart") then p.Anchored, p.CanCollide = false, false end end
            local pr = wingsModel.PrimaryPart or wingsModel:FindFirstChildWhichIsA("BasePart", true)
            if pr then
                wingsModel.PrimaryPart = pr
                local wc = Instance.new("WeldConstraint")
                wc.Part0, wc.Part1, wc.Parent = torso, pr, pr
            end
        end
        local function loadHalo(id)
            local s, o = pcall(function() return game:GetObjects("rbxassetid://"..id) end)
            if s and o and #o > 0 then
                local h = o[1]
                if not h:IsA("Model") then
                    local m = Instance.new("Model")
                    for _, x in ipairs(o) do x.Parent = m end
                    h = m
                end
                h.Parent = chr
                for _, p in ipairs(h:GetDescendants()) do if p:IsA("BasePart") then p.Anchored, p.CanCollide = false, false end end
                local pr = h.PrimaryPart or h:FindFirstChildWhichIsA("BasePart", true)
                if pr then
                    h.PrimaryPart = pr
                    local wc = Instance.new("WeldConstraint")
                    wc.Part0, wc.Part1, wc.Parent = head, pr, pr
                end
                return h
            end
        end
        halo1Model = loadHalo(HALO1_ID) halo2Model = loadHalo(HALO2_ID)
        local bv = Instance.new("BodyVelocity", hrp)
        bv.Name, bv.MaxForce, bv.Velocity = "FlyVelocity", Vector3.new(1e9,1e9,1e9), Vector3.zero
        local bg = Instance.new("BodyGyro", hrp)
        bg.Name, bg.MaxTorque, bg.CFrame = "FlyGyro", Vector3.new(1e9,1e9,1e9), hrp.CFrame
        if hum then hum.PlatformStand = true end
        local an = hum and hum:FindFirstChild("Animator")
        local ia = Instance.new("Animation") ia.AnimationId = FLY_IDLE
        local ma = Instance.new("Animation") ma.AnimationId = FLY_MOVE
        local it = an and an:LoadAnimation(ia)
        local mt = an and an:LoadAnimation(ma)
        if it then it:Play() end
        local lastM, cam = false, workspace.CurrentCamera
        local wt, ht = 0, 0
        flyConnections.RS = RunService.RenderStepped:Connect(function(dt)
            if not hrp or not hrp.Parent then return end
            local dir = Vector3.zero
            local ccf = cam.CFrame
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + ccf.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - ccf.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - ccf.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + ccf.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0,1,0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0,1,0) end
            if hum and hum.MoveDirection.Magnitude > 0 then dir = dir + hum.MoveDirection end
            if dir.Magnitude > 0 then
                bv.Velocity = dir.Unit * 55
                if not lastM then lastM = true if it then it:Stop() end if mt and not mt.IsPlaying then mt:Play() end end
            else
                bv.Velocity = Vector3.zero
                if lastM then lastM = false if mt then mt:Stop() end if it and not it.IsPlaying then it:Play() end end
            end
            bg.CFrame = ccf
            wt = wt + dt * 5
            if wingsModel and wingsModel.PrimaryPart then
                wingsModel:SetPrimaryPartCFrame(torso.CFrame * CFrame.new(0,-0.2,0.6) * CFrame.Angles(0, math.pi, math.sin(wt)*0.15))
            end
            ht = ht + dt
            local r = tick() * 3.5
            if halo1Model and halo1Model.PrimaryPart then halo1Model:SetPrimaryPartCFrame(head.CFrame * CFrame.new(0,1.3,0) * CFrame.Angles(0, r, 0)) end
            if halo2Model and halo2Model.PrimaryPart then halo2Model:SetPrimaryPartCFrame(head.CFrame * CFrame.new(0,1.3,0) * CFrame.Angles(0, -r, 0)) end
            local t = math.floor(ht*2) % 2 == 0 and 0 or 1
            if halo1Model then for _, p in ipairs(halo1Model:GetDescendants()) do if p:IsA("BasePart") then p.Transparency = t end end end
            if halo2Model then for _, p in ipairs(halo2Model:GetDescendants()) do if p:IsA("BasePart") then p.Transparency = 1-t end end end
        end)
    else
        for _, c in pairs(flyConnections) do c:Disconnect() end
        flyConnections = {}
        if wingsModel then wingsModel:Destroy() wingsModel = nil end
        if halo1Model then halo1Model:Destroy() halo1Model = nil end
        if halo2Model then halo2Model:Destroy() halo2Model = nil end
        if hrp then
            if hrp:FindFirstChild("FlyVelocity") then hrp.FlyVelocity:Destroy() end
            if hrp:FindFirstChild("FlyGyro") then hrp.FlyGyro:Destroy() end
        end
        if hum then hum.PlatformStand = false end
    end
end
local function spawnSkyline(targetName)
    local tp = Players:FindFirstChild(targetName)
    if not tp or not tp.Character then notify("ADM", "Alvo inválido!") return end
    local th = tp.Character:FindFirstChild("HumanoidRootPart") if not th then return end
    task.spawn(function()
        notify("ADM", "Skyline → "..targetName)
        local ok, objs = pcall(function() return game:GetObjects("rbxassetid://"..SKYLINE_MODEL_ID) end)
        if not (ok and objs and #objs > 0) then return end
        local car = objs[1]
        if not car:IsA("Model") then
            local m = Instance.new("Model")
            for _, o in ipairs(objs) do o.Parent = m end
            car = m
        end
        if not car.PrimaryPart then car.PrimaryPart = car:FindFirstChildWhichIsA("BasePart", true) end
        car.Parent = workspace
        pcall(function() car:ScaleTo(0.35) end)
        for _, p in ipairs(car:GetDescendants()) do if p:IsA("BasePart") then p.Anchored, p.CanCollide = true, false end end
        local sp = th.Position + th.CFrame.LookVector*45
        local rp = RaycastParams.new()
        rp.FilterType, rp.FilterDescendantsInstances = Enum.RaycastFilterType.Exclude, {car, tp.Character}
        local rr = workspace:Raycast(sp + Vector3.new(0,50,0), Vector3.new(0,-100,0), rp)
        local gy = rr and rr.Position.Y or sp.Y
        local bcf, bs = car:GetBoundingBox()
        local bo = car:GetPivot().Position.Y - (bcf.Position.Y - bs.Y/2)
        local tcf = CFrame.new(sp.X, gy+bo, sp.Z) * CFrame.lookAt(sp, th.Position).Rotation
        local scf = tcf * CFrame.new(0,-10,0)
        car:PivotTo(scf)
        for i = 1, 35 do car:PivotTo(scf:Lerp(tcf, (i/35)^2*(3-2*(i/35)))) task.wait(0.015) end
        car:PivotTo(tcf) task.wait(0.15)
        local st, dur, hit = tick(), 10, false
        while tick()-st < dur and car and car.Parent do
            if tp and tp.Character then
                local cth = tp.Character:FindFirstChild("HumanoidRootPart")
                local chh = tp.Character:FindFirstChildOfClass("Humanoid")
                if cth then
                    local ccf = car:GetPivot()
                    local cr = workspace:Raycast(cth.Position + Vector3.new(0,20,0), Vector3.new(0,-40,0), rp)
                    local cgy = cr and cr.Position.Y or cth.Position.Y
                    local tpos = Vector3.new(cth.Position.X, cgy+bo, cth.Position.Z)
                    local d = tpos - ccf.Position
                    if d.Magnitude > 2 then car:PivotTo(CFrame.new(ccf.Position + d.Unit*3.75, tpos)) end
                    if (ccf.Position - cth.Position).Magnitude <= 7 and not hit then
                        hit = true
                        if chh then chh.Health = 0 end
                        Instance.new("Explosion", workspace).Position = cth.Position
                    end
                end
            else break end
            task.wait(0.03)
        end
        task.wait(0.4) if car then car:Destroy() end
    end)
end

    local function findCouch()
        local chr = LocalPlayer.Character
        for _, c in ipairs({ LocalPlayer:FindFirstChild("Backpack"), chr }) do
            if c then
                for _, t in ipairs(c:GetChildren()) do
                    if t:IsA("Tool") then
                        local n = t.Name:lower()
                        if t.Name == "Couch" or t.Name == "Chaos.Couch" or n:find("couch") or n:find("sofa") then return t end
                    end
                end
            end
        end
    end
    local function KillPlayerCouch()
        local targetName = getgenv().Target or selectedPlayerNameBH
        local tp = targetName and Players:FindFirstChild(targetName)
        if not tp or not tp.Character or not tp.Character:FindFirstChild("HumanoidRootPart") then notify("Couch", "Selecione um jogador.") return end
        local RE = ReplicatedStorage:FindFirstChild("RE")
        if not RE then notify("Couch", "RE não encontrado.") return end
        pcall(function() RE:FindFirstChild("1ClearToolis"):FireServer("ClearAllTools") end)
        task.wait(0.4)
        local couch
        for _ = 1, 5 do
            pcall(function() RE:FindFirstChild("1Tool"):InvokeServer("PickingTools", "Couch") end)
            local t0 = tick()
            repeat couch = findCouch() task.wait(0.1) until couch or tick()-t0 > 1.5
            if couch then break end
        end
        if not couch then notify("Couch", "Sofá não encontrado.") return end
        couch.Name = "Chaos.Couch"
        local s1 = couch:FindFirstChild("Seat1") or couch:FindFirstChildWhichIsA("Seat", true)
        local s2 = couch:FindFirstChild("Seat2") or s1
        local hd = couch:FindFirstChild("Handle")
        if not s1 then notify("Couch", "Sofá sem assento.") return end
        if s1 and s2 and hd then s1.Disabled, s2.Disabled, hd.Name = true, true, "Handle " end
        couch.Parent = LocalPlayer.Character
        local bv = Instance.new("BodyVelocity", s1)
        bv.MaxForce, bv.P, bv.Velocity = Vector3.new(1e9,1e9,1e9), 1250, Vector3.zero
        bv.Name = "#MOVODEPFS#@F$GERE..>V<<<EwV<<W"
        local t0 = tick()
        repeat
            local tr = tp.Character and tp.Character:FindFirstChild("HumanoidRootPart")
            if not tr then break end
            s1.CFrame = CFrame.new(tr.Position + tr.Velocity*0.5)*CFrame.new(-2,2,0)
            task.wait()
        until (tp.Character and tp.Character:FindFirstChild("Humanoid") and tp.Character.Humanoid.Sit) or tick()-t0 > 8
        task.wait()
        couch.Parent = LocalPlayer.Backpack
        s1.CFrame = CFrame.new(9999,-450,9999)
        s2.CFrame = CFrame.new(9999,-450,9999)
        couch.Parent = LocalPlayer.Character
        task.wait(0.1)
        couch.Parent = LocalPlayer.Backpack
        task.wait(2)
        local b = s1:FindFirstChild("#MOVODEPFS#@F$GERE..>V<<<EwV<<W")
        if b then b:Destroy() end
        pcall(function() RE:FindFirstChild("1ClearToolis"):FireServer("ClearAllTools") end)
    end
    local function KillPlayerBus()
        local chr = LocalPlayer.Character
        local hrp = chr and chr:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        local orig = hrp.CFrame
        local function GetBus() local v = workspace:FindFirstChild("Vehicles") return v and v:FindFirstChild(LocalPlayer.Name .. "Car") end
        local bus = GetBus()
        if not bus then
            hrp.CFrame = CFrame.new(1118.81, 75.998, -1138.61)
            task.wait(0.5)
            local re = ReplicatedStorage:FindFirstChild("RE")
            if re and re:FindFirstChild("1Cair") then re["1Cair"]:FireServer("PickingCar", "SchoolBus") end
            task.wait(1) bus = GetBus()
        end
        if bus then
            local seat = bus:FindFirstChild("Body") and bus.Body:FindFirstChild("VehicleSeat")
            if seat and chr:FindFirstChildOfClass("Humanoid") and not chr.Humanoid.Sit then
                repeat hrp.CFrame = seat.CFrame * CFrame.new(0,2,0) task.wait() until chr.Humanoid.Sit or not bus.Parent
            end
            local tp = Players:FindFirstChild(getgenv().Target or selectedPlayerNameBH)
            if tp and tp.Character and tp.Character:FindFirstChild("HumanoidRootPart") then
                local tr = tp.Character.HumanoidRootPart
                for _ = 1, 60 do
                    if not tp.Character or not tp.Character:FindFirstChild("HumanoidRootPart") then break end
                    local tt = tick()*35
                    pcall(function() bus:SetPrimaryPartCFrame(tr.CFrame * CFrame.new(math.sin(tt)*4, 0, math.cos(tt)*20)) end)
                    RunService.RenderStepped:Wait()
                end
            end
            hrp.CFrame = orig
            pcall(function() ReplicatedStorage.RE["1Cair"]:FireServer("DeleteAllVehicles") end)
        end
    end
