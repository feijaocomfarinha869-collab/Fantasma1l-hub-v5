local function getRealImageIdDelta(id)
    local cleanId = tostring(id):match("%d+")
    if not cleanId then return "rbxassetid://" .. tostring(id) end
    return "rbxassetid://" .. cleanId
end

local function safePlayAnimation(character, animId)
    if not character then return nil end
    local hum = character:FindFirstChild("Humanoid")
    local anim = hum and hum:FindFirstChild("Animator")
    if not anim then return nil end
    local a = Instance.new("Animation")
    a.AnimationId = animId
    local ok, track = pcall(function() return anim:LoadAnimation(a) end)
    if ok and track then track:Play() return track end
    return nil
end

local function fetchAssetAsModel(id)
    local s, objs = pcall(function() return game:GetObjects("rbxassetid://" .. tostring(id)) end)
    if s and objs and #objs > 0 then
        local m = Instance.new("Model")
        for _, o in ipairs(objs) do o.Parent = m end
        if not m.PrimaryPart then m.PrimaryPart = m:FindFirstChildWhichIsA("BasePart", true) end
        return m
    end
    return nil
end

local function makeModelFallSlowly(model)
    task.spawn(function()
        while model and model.Parent do
            for _, p in ipairs(model:GetDescendants()) do
                if p:IsA("BasePart") and not p.Anchored then
                    p.AssemblyLinearVelocity = Vector3.new(0, -7, 0)
                end
            end
            task.wait(0.1)
        end
    end)
end

local function sendNetCmd(cmd)
    pcall(function()
        if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
            local ch = TextChatService:FindFirstChild("TextChannels")
                and TextChatService.TextChannels:FindFirstChild("RBXGeneral")
            if ch then ch:SendAsync(cmd) end
        else
            local sayReq = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
                and ReplicatedStorage.DefaultChatSystemChatEvents:FindFirstChild("SayMessageRequest")
            if sayReq then sayReq:FireServer(cmd, "All") end
        end
    end)
end

function applyOverheadTag(player)
    pcall(function()
        if not player or not player.Character then return end
        local head = player.Character:WaitForChild("Head", 3)
        if not head then return end
        if head:FindFirstChild("FantasmaOverhead") then head.FantasmaOverhead:Destroy() end

        local cargo = "USUARIO"
        if WHITELIST[player.Name] then cargo = WHITELIST[player.Name].Role
        elseif NETWORK_ROLES[player.Name] then cargo = NETWORK_ROLES[player.Name] end
        if player == LocalPlayer then cargo = cargo .. " (VOCÊ)" end

        local bb = Instance.new("BillboardGui", head)
        bb.Name, bb.Size, bb.StudsOffset, bb.AlwaysOnTop = "FantasmaOverhead", UDim2.new(0, 240, 0, 50), Vector3.new(0, 3, 0), true

        local fr = Instance.new("Frame", bb)

        fr.BackgroundTransparency, fr.BorderSizePixel = 0.25, 0
        Instance.new("UICorner", fr).CornerRadius = UDim.new(0, 8)

        local st = Instance.new("UIStroke", fr)
        st.Color, st.Thickness = Color3.fromRGB(168, 85, 247), 1.8

        local ic = Instance.new("ImageLabel", fr)
        ic.Size, ic.Position, ic.BackgroundTransparency, ic.Image = UDim2.new(0, 32, 0, 32), UDim2.new(0, 8, 0.5, -16), 1, ICON_ID

        local lbl = Instance.new("TextLabel", fr)
        lbl.Size, lbl.Position, lbl.BackgroundTransparency = UDim2.new(1, -50, 1, 0), UDim2.new(0, 45, 0, 0), 1

        lbl.Font, lbl.TextXAlignment = Enum.Font.GothamBold, Enum.TextXAlignment.Left
    end)

-- ================== KAMUI ==================
local function toggleKamuiDimensionLocal(state)
    kamuiEnabled = state
    if kamuiEnabled then
        Lighting.FogColor = Color3.fromRGB(0, 0, 0)
        Lighting.FogStart, Lighting.FogEnd = 0, 300
        Lighting.Ambient, Lighting.OutdoorAmbient = Color3.fromRGB(15, 15, 20), Color3.fromRGB(15, 15, 20)

        kamuiFolder = Instance.new("Folder", workspace)
        kamuiFolder.Name = "KamuiDimensionBox"

        local c = Vector3.new(0, 500, 0)
        local hx, hy, hz = kamuiSize.X / 2, kamuiSize.Y / 2, kamuiSize.Z / 2
        local wt = 12

        local function wall(sz, pos)
            local p = Instance.new("Part", kamuiFolder)
            p.Size, p.Position, p.Anchored, p.CanCollide, p.CastShadow = sz, c + pos, true, true, false
            p.Material, p.Color = Enum.Material.SmoothPlastic, Color3.fromRGB(10, 10, 10)
        end
        wall(Vector3.new(kamuiSize.X, wt, kamuiSize.Z), Vector3.new(0, -hy, 0))
        wall(Vector3.new(kamuiSize.X, wt, kamuiSize.Z), Vector3.new(0, hy, 0))
        wall(Vector3.new(wt, kamuiSize.Y, kamuiSize.Z), Vector3.new(-hx, 0, 0))
        wall(Vector3.new(wt, kamuiSize.Y, kamuiSize.Z), Vector3.new(hx, 0, 0))
        wall(Vector3.new(kamuiSize.X, kamuiSize.Y, wt), Vector3.new(0, 0, -hz))
        wall(Vector3.new(kamuiSize.X, kamuiSize.Y, wt), Vector3.new(0, 0, hz))

        local cx, cz = hx - 15, hz - 15
        local lh = kamuiSize.Y - 40
        for _, pos in ipairs({Vector3.new(-cx,0,-cz), Vector3.new(cx,0,-cz), Vector3.new(-cx,0,cz), Vector3.new(cx,0,cz)}) do
            local l = Instance.new("Part", kamuiFolder)
            l.Size, l.Position, l.Anchored, l.CanCollide = Vector3.new(6, lh, 6), c + pos, true, false
            l.Material, l.Color = Enum.Material.Neon, Color3.fromRGB(200, 220, 255)
            local pl = Instance.new("PointLight", l)
            pl.Color, pl.Range, pl.Brightness = Color3.fromRGB(200, 220, 255), 250, 3
        end

        local bs, sp, gr = 18, 6, 12
        local fy = c.Y - hy + wt / 2
        for x = -gr, gr do
            for z = -gr, gr do
                if math.abs(x) > 2 or math.abs(z) > 2 then
                    local h = math.random(25, 140)
                    local b = Instance.new("Part", kamuiFolder)
                    b.Size = Vector3.new(bs, h, bs)
                    b.Position = Vector3.new(c.X + x * (bs + sp), fy + h / 2, c.Z + z * (bs + sp))
                    b.Anchored, b.Material, b.Color = true, Enum.Material.Concrete, Color3.fromRGB(115, 125, 135)
                end
            end
        end

        local pf = Instance.new("Part", kamuiFolder)
        pf.Name, pf.Size = "KamuiTargetPlatform", Vector3.new(70, 4, 70)
        pf.Position, pf.Anchored = Vector3.new(c.X, fy + 85, c.Z), true
        pf.Material, pf.Color = Enum.Material.Slate, Color3.fromRGB(190, 200, 210)

        task.spawn(function()
            task.wait(0.1)
            local hrp = getHRP()
            if hrp then hrp.CFrame = pf.CFrame + Vector3.new(0, 6, 0) end
        end)
        notify("Kamui", "Dimensão gerada!")
    else
        if kamuiFolder then kamuiFolder:Destroy() kamuiFolder = nil end
        Lighting.FogEnd = 100000
        notify("Kamui", "Dimensão fechada!")
    end
end

function executeActionLocally(action, senderName, extraArg)
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")

    if action == "KILL" and hum then hum.Health = 0
    elseif action == "KICK" then LocalPlayer:Kick("Expulso pelo Fantasma1l Hub.")
    elseif action == "FREEZE" and hrp and hum then hrp.Anchored, hum.WalkSpeed, hum.JumpPower = true, 0, 0
    elseif action == "UNFREEZE" and hrp and hum then hrp.Anchored, hum.WalkSpeed, hum.JumpPower = false, 16, 50
    elseif action == "CRASH" then while true do end
    elseif action == "EXPLODE" and hrp then
        Instance.new("Explosion", workspace).Position = hrp.Position
        if hum then hum.Health = 0 end
    elseif action == "ICE" and hrp then
        if workspace:FindFirstChild("Ice_" .. LocalPlayer.Name) then workspace["Ice_" .. LocalPlayer.Name]:Destroy() end
        local ip = Instance.new("Part", workspace)
        ip.Name, ip.Size, ip.CFrame = "Ice_" .. LocalPlayer.Name, Vector3.new(6, 8, 6), hrp.CFrame
        ip.Anchored, ip.CanCollide = true, false
        ip.Color, ip.Material, ip.Transparency = Color3.fromRGB(0, 170, 255), Enum.Material.Glass, 0.4
        if hum then hum.WalkSpeed = 0 end
        hrp.Anchored = true
    elseif action == "UNICE" and hrp then
        if workspace:FindFirstChild("Ice_" .. LocalPlayer.Name) then workspace["Ice_" .. LocalPlayer.Name]:Destroy() end
        if hum then hum.WalkSpeed = 16 end
        hrp.Anchored = false
    elseif action == "JAIL" and extraArg then
        local targetPlr = Players:FindFirstChild(extraArg)
        if targetPlr and targetPlr.Character and targetPlr.Character:FindFirstChild("HumanoidRootPart") then
            local thrp = targetPlr.Character.HumanoidRootPart
            if workspace:FindFirstChild("Jail_" .. targetPlr.Name) then
                workspace["Jail_" .. targetPlr.Name]:Destroy()
            end
            local cage = Instance.new("Model", workspace)
            cage.Name = "Jail_" .. targetPlr.Name

                {Vector3.new(6,1,6), cf * CFrame.new(0,-3.5,0)},
                {Vector3.new(6,1,6), cf * CFrame.new(0, 3.5,0)},
                {Vector3.new(1,6,6), cf * CFrame.new( 3,0,0)},
                {Vector3.new(1,6,6), cf * CFrame.new(-3,0,0)},
                {Vector3.new(6,6,1), cf * CFrame.new(0,0, 3)},
                {Vector3.new(6,6,1), cf * CFrame.new(0,0,-3)},
            }) do
                local p = Instance.new("Part", cage)
                p.Size, p.CFrame = d[1], d[2]

local recentNet = {}
local function isMe(t)
    if not t then return false end
    local l = string.lower(t)
    return l == string.lower(LocalPlayer.Name) or l == string.lower(LocalPlayer.DisplayName)
end

function processNetMessage(msg, senderName)
    if not msg then return end
    do
        local now = os.clock()
        local key = tostring(senderName) .. ":" .. msg
        if recentNet[key] and now - recentNet[key] < 1.5 then return end
        recentNet[key] = now
        for k, t in pairs(recentNet) do
            if now - t > 5 then recentNet[k] = nil end
        end
    end
    local body
    if msg:sub(1, 1) == ";" then body = msg:sub(2)
    elseif string.lower(msg:sub(1, 5)) == "fhub " then body = msg:sub(6)
    else return end

    local parts = string.split(body, " ")
    if #parts < 1 or parts[1] == "" then return end

    local action = string.upper(parts[1])
    local target = parts[2]
    local extraArg = parts[3] and table.concat(parts, " ", 3) or nil

    if action == "PING" then
        local role = (WHITELIST[LocalPlayer.Name] and WHITELIST[LocalPlayer.Name].Role) or "USER"
        sendNetCmd(";PONG " .. LocalPlayer.Name .. " ROLE_" .. role)
    elseif action == "PONG" and target then
        activeScriptUsers[target] = true
        local roleExtra = parts[3] or "ROLE_USER"
        NETWORK_ROLES[target] = roleExtra:gsub("ROLE_", "")
        local pt = Players:FindFirstChild(target)
        if pt and IS_ADMIN then applyOverheadTag(pt) end
    end

    if action == "LOCUST_STATE" then
        if target and senderName == target then LOCUST.setUser(target, extraArg == "ON") end
        return
    end
    if action == "XESTER_STATE" then
        if target and senderName == target then XESTER.setUser(target, extraArg == "ON") end
        return
    end

    if action ~= "PING" and action ~= "PONG" then
        if REQUIRE_WHITELIST and senderName ~= LocalPlayer.Name and not WHITELIST[senderName] then return end
    end

    if FXNET.actions[action] then
        if target then
            task.spawn(function() pcall(executeActionLocally, action, senderName, target) end)
        end
        return
    end

    if action == "KAMUI" or action == "COOLKID_EFFECT" then
        if target == "ALL" or target == LocalPlayer.Name or senderName == LocalPlayer.Name or target == selectedPlayerName then
            task.spawn(function() pcall(function() executeActionLocally(action, senderName, target or extraArg) end) end)
        end
    else
        if target and (string.upper(target) == "ALL" or isMe(target)) then
            task.spawn(function() pcall(function() executeActionLocally(action, senderName, extraArg) end) end)
        end
    end
end

local BLACKHOLE = { active = false, conn = nil, deathConn = nil, charConn = nil, hole = nil, saved = {} }

local function stopBlackhole(silent)
    if not BLACKHOLE.active then return end
    BLACKHOLE.active = false
    if BLACKHOLE.conn then pcall(function() BLACKHOLE.conn:Disconnect() end) end
    if BLACKHOLE.deathConn then pcall(function() BLACKHOLE.deathConn:Disconnect() end) end
    if BLACKHOLE.charConn then pcall(function() BLACKHOLE.charConn:Disconnect() end) end
    BLACKHOLE.conn, BLACKHOLE.deathConn, BLACKHOLE.charConn = nil, nil, nil
    local n = 0
    for p, s in pairs(BLACKHOLE.saved) do
        if p and p.Parent then
            pcall(function()
                p.AssemblyLinearVelocity = s.vel
                p.AssemblyAngularVelocity = s.avel
            end)
            n += 1
        end
    end
    BLACKHOLE.saved = {}
    if BLACKHOLE.hole then pcall(function() BLACKHOLE.hole:Destroy() end) BLACKHOLE.hole = nil end
    if not silent then notify("Blackhole", n .. " objetos voltaram ao normal.") end
end

local function startBlackhole(posStr)
    if BLACKHOLE.active then stopBlackhole(true) end
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local pos
    if posStr then
        local x, y, z = posStr:match("([%-%d%.]+),([%-%d%.]+),([%-%d%.]+)")
        if x then pos = Vector3.new(tonumber(x), tonumber(y), tonumber(z)) end
    end
    if not pos then
        pos = hrp and (hrp.Position + hrp.CFrame.LookVector * 140 + Vector3.new(0, 70, 0)) or Vector3.new(0, 220, 0)
    end
    BLACKHOLE.active = true

    local hole = Instance.new("Part")
    hole.Name = "F1L_Blackhole"
    hole.Shape = Enum.PartType.Ball
    hole.Size = Vector3.new(28, 28, 28)
    hole.Position = pos
    hole.Anchored = true
    hole.CanCollide = false
    hole.CanTouch = false
    hole.CanQuery = false
    hole.Material = Enum.Material.Neon
    hole.Color = Color3.fromRGB(0, 0, 0)
    hole.Parent = workspace

    local glow = Instance.new("PointLight")
    glow.Color = Color3.fromRGB(90, 30, 180)
    glow.Range = 200
    glow.Brightness = 4
    glow.Parent = hole

    local acc = Instance.new("ParticleEmitter")
    acc.Texture = "rbxasset://textures/particles/sparkles_main.dds"
    acc.Color = ColorSequence.new(PURPLE, BLUE)
    acc.Size = NumberSequence.new({ NumberSequenceKeypoint.new(0, 3), NumberSequenceKeypoint.new(1, 0) })
    acc.Transparency = NumberSequence.new({ NumberSequenceKeypoint.new(0, 0), NumberSequenceKeypoint.new(1, 1) })
    acc.Lifetime = NumberRange.new(0.6, 1.2)
    acc.Rate = 90
    acc.Speed = NumberRange.new(30, 60)
    acc.SpreadAngle = Vector2.new(360, 360)
    acc.LightEmission = 1
    acc.Parent = hole

    BLACKHOLE.hole = hole
    BLACKHOLE.saved = {}

    local ex = { hole, char }
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then table.insert(ex, p.Character) end
    end
    local params = OverlapParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = ex
    params.MaxParts = 900
    local found = workspace:GetPartBoundsInRadius(pos, 4500, params)
    local count = 0
    for _, p in ipairs(found) do
        if p:IsA("BasePart") and not p.Anchored and not p:IsA("Terrain") then
            if not BLACKHOLE.saved[p] then
                BLACKHOLE.saved[p] = {
                    vel = p.AssemblyLinearVelocity,
                    avel = p.AssemblyAngularVelocity,
                }
                count += 1
            end
        end
    end
    notify("Blackhole", count .. " objetos sendo sugados!")

    BLACKHOLE.conn = RunService.Heartbeat:Connect(function(dt)
        if not BLACKHOLE.active then return end
        hole.CFrame = hole.CFrame * CFrame.Angles(0, dt * 3, dt)
        for p, s in pairs(BLACKHOLE.saved) do
            if p and p.Parent then
                local d = hole.Position - p.Position
                local dist = math.max(d.Magnitude, 4)
                if dist > hole.Size.X / 2 + 5 then
                    local pull = math.clamp(220000 / (dist * dist), 30, 600)
                    p.AssemblyLinearVelocity = d.Unit * pull + Vector3.new(0, dist * 0.10, 0)
                    p.AssemblyAngularVelocity = Vector3.new(0, 12, 0)
                else
                    p.CFrame = hole.CFrame
                    p.AssemblyLinearVelocity = Vector3.zero
                    p.AssemblyAngularVelocity = Vector3.zero
                end
            else
                BLACKHOLE.saved[p] = nil
            end
        end
    end)

    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then
        BLACKHOLE.deathConn = hum.Died:Connect(function()
            task.delay(2, function() pcall(stopBlackhole) end)
        end)
    end
    BLACKHOLE.charConn = LocalPlayer.CharacterAdded:Connect(function()
        task.delay(2, function() pcall(stopBlackhole) end)
    end)
end

        end

    end
end

                    local cgy = cr and cr.Position.Y or cth.Position.Y
                    local tpos = Vector3.new(cth.Position.X, cgy + bo, cth.Position.Z)
                    local d = tpos - ccf.Position
                    if d.Magnitude > 2 then car:PivotTo(CFrame.new(ccf.Position + d.Unit * 3.75, tpos)) end
                    if (ccf.Position - cth.Position).Magnitude <= 7 and not hit then
                        hit = true
                        if ch then ch.Health = 0 end
                        Instance.new("Explosion", workspace).Position = cth.Position
                    end
                end
            else break end
            task.wait(0.03)
        end
        task.wait(0.4)
        if car then car:Destroy() end
    end)
end

local function toggleFlyWithWings()
    flyWingsActive = not flyWingsActive
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local torso = char and (char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso"))
    local head = char and char:FindFirstChild("Head")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not char or not hrp or not torso or not head then return end
    if flyWingsActive then
        local ok, w = pcall(function() return game:GetObjects("rbxassetid://" .. WINGS_ID) end)
        if ok and w and #w > 0 then
            wingsModel = w[1]
            if not wingsModel:IsA("Model") then
                local m = Instance.new("Model")
                for _, o in ipairs(w) do o.Parent = m end
                wingsModel = m
            end
            wingsModel.Parent = char
            for _, p in ipairs(wingsModel:GetDescendants()) do
                if p:IsA("BasePart") then p.Anchored, p.CanCollide = false, false end
            end
            local pr = wingsModel.PrimaryPart or wingsModel:FindFirstChildWhichIsA("BasePart", true)
            if pr then
                wingsModel.PrimaryPart = pr
                local wc = Instance.new("WeldConstraint")
                wc.Part0, wc.Part1, wc.Parent = torso, pr, pr
            end
        end
        local function loadHalo(id)
            local s, o = pcall(function() return game:GetObjects("rbxassetid://" .. id) end)
            if s and o and #o > 0 then
                local h = o[1]
                if not h:IsA("Model") then
                    local m = Instance.new("Model")
                    for _, x in ipairs(o) do x.Parent = m end
                    h = m
                end
                h.Parent = char
                for _, p in ipairs(h:GetDescendants()) do
                    if p:IsA("BasePart") then p.Anchored, p.CanCollide = false, false end
                end
                local pr = h.PrimaryPart or h:FindFirstChildWhichIsA("BasePart", true)
                if pr then
                    h.PrimaryPart = pr
                    local wc = Instance.new("WeldConstraint")
                    wc.Part0, wc.Part1, wc.Parent = head, pr, pr
                end
                return h
            end
        end
        halo1Model = loadHalo(HALO1_ID)
        halo2Model = loadHalo(HALO2_ID)
        local bv = Instance.new("BodyVelocity", hrp)
        bv.Name, bv.MaxForce, bv.Velocity = "FlyVelocity", Vector3.new(1e9,1e9,1e9), Vector3.zero
        local bg = Instance.new("BodyGyro", hrp)
        bg.Name, bg.MaxTorque, bg.CFrame = "FlyGyro", Vector3.new(1e9,1e9,1e9), hrp.CFrame
        if hum then hum.PlatformStand = true end
        local an = hum and hum:FindFirstChild("Animator")
        local ia = Instance.new("Animation") ia.AnimationId = FLY_IDLE
        local ma = Instance.new("Animation") ma.AnimationId = FLY_MOVE
        local it = an and an:LoadAnimation(ia)
        local mt = an and an:LoadAnimation(ma)
        if it then it:Play() end
        local lastM, cam = false, workspace.CurrentCamera
        local wt, ht = 0, 0
        flyConnections.RS = RunService.RenderStepped:Connect(function(dt)
            if not hrp or not hrp.Parent then return end
            local dir = Vector3.zero
            local ccf = cam.CFrame
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + ccf.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - ccf.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - ccf.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + ccf.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0,1,0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0,1,0) end
            if hum and hum.MoveDirection.Magnitude > 0 then dir = dir + hum.MoveDirection end
            if dir.Magnitude > 0 then
                bv.Velocity = dir.Unit * 55
                if not lastM then
                    lastM = true
                    if it then it:Stop() end
                    if mt and not mt.IsPlaying then mt:Play() end
                end
            else
                bv.Velocity = Vector3.zero
                if lastM then
                    lastM = false
                    if mt then mt:Stop() end
                    if it and not it.IsPlaying then it:Play() end
                end
            end
            bg.CFrame = ccf
            wt = wt + dt * 5
            if wingsModel and wingsModel.PrimaryPart then
                wingsModel:SetPrimaryPartCFrame(torso.CFrame * CFrame.new(0,-0.2,0.6) * CFrame.Angles(0, math.pi, math.sin(wt)*0.15))
            end
            ht = ht + dt
            local r = tick() * 3.5
            if halo1Model and halo1Model.PrimaryPart then
                halo1Model:SetPrimaryPartCFrame(head.CFrame * CFrame.new(0,1.3,0) * CFrame.Angles(0, r, 0))
            end
            if halo2Model and halo2Model.PrimaryPart then
                halo2Model:SetPrimaryPartCFrame(head.CFrame * CFrame.new(0,1.3,0) * CFrame.Angles(0, -r, 0))
            end
            local t = math.floor(ht * 2) % 2 == 0 and 0 or 1
            if halo1Model then for _, p in ipairs(halo1Model:GetDescendants()) do if p:IsA("BasePart") then p.Transparency = t end end end
            if halo2Model then for _, p in ipairs(halo2Model:GetDescendants()) do if p:IsA("BasePart") then p.Transparency = 1 - t end end end
        end)
    else
        for _, c in pairs(flyConnections) do c:Disconnect() end
        flyConnections = {}
        if wingsModel then wingsModel:Destroy() wingsModel = nil end
        if halo1Model then halo1Model:Destroy() halo1Model = nil end
        if halo2Model then halo2Model:Destroy() halo2Model = nil end
        if hrp then
            if hrp:FindFirstChild("FlyVelocity") then hrp.FlyVelocity:Destroy() end
            if hrp:FindFirstChild("FlyGyro") then hrp.FlyGyro:Destroy() end
        end
        if hum then hum.PlatformStand = false end
    end
end

local function triggerCommand(action, target, extraArg)
    if not target or target == "" then return end
    if target == LocalPlayer.Name or target == "ALL" then
        executeActionLocally(action, LocalPlayer.Name, extraArg)
    end
    local cmd = ";" .. action .. " " .. target
    if extraArg then cmd = cmd .. " " .. extraArg end
    sendNetCmd(cmd)
    notify("ADM", action .. " → " .. target)
end

task.spawn(function()
    local ok, redz = pcall(function()
        return loadstring(game:HttpGet("https://raw.githubusercontent.com/minhdepzai-v/LibraryRobloc/refs/heads/main/RedzLibrary.lua"))()
    end)
    if not (ok and redz) then return end

    local Window = redz:MakeWindow({
        Title = "Fantasma1l Hub [" .. MY_ROLE .. "]",
        SubTitle = "Brookhaven Edition",
        SaveFolder = "Fantasma1l_Brookhaven"
    })
    Window:AddMinimizeButton({
        Button = { Image = ICON_ID, BackgroundTransparency = 0 },
        Corner = { CornerRadius = UDim.new(35, 1) },
    })

    local TabHome = Window:MakeTab({ "Início", "home" })
    local TabTroll = Window:MakeTab({ "Troll & Ações", "user" })
    local TabAudio = Window:MakeTab({ "Áudio", "music" })
    Window:SelectTab(TabHome)

    TabHome:AddSection({ "Informações" })
    TabHome:AddParagraph({ "Bem-vindo", "Cargo: " .. MY_ROLE })
    TabHome:AddButton({ "Rejoin Server", function()
        game:GetService("TeleportService"):TeleportToPlaceInstance(game.PlaceId, game.JobId, LocalPlayer)
    end })

    TabTroll:AddSection({ "Troll Player" })
    local selectedPlayerNameBH = nil
    getgenv().Target = nil
    local function listPlayers()
        local names = {}
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then table.insert(names, p.Name) end
        end
        if #names == 0 then table.insert(names, "Ninguém") end
        return names
    end
    local dd = TabTroll:AddDropdown({ Name = "Selecionar Jogador", Options = listPlayers(), Default = "",
        Callback = function(v) selectedPlayerNameBH = v getgenv().Target = v end })
    TabTroll:AddButton({ "Atualizar Lista", function() dd:Set(listPlayers()) end })
    TabTroll:AddButton({ "Teleportar até Player", function()
        if not selectedPlayerNameBH then return end
        local tp = Players:FindFirstChild(selectedPlayerNameBH)
        local chr = LocalPlayer.Character
        local h = chr and chr:FindFirstChild("HumanoidRootPart")
        if tp and tp.Character and tp.Character:FindFirstChild("HumanoidRootPart") and h then
            h.CFrame = tp.Character.HumanoidRootPart.CFrame
        end
    end })
    local specConn
    TabTroll:AddToggle({ Name = "Spectar Player", Default = false, Callback = function(v)
        if specConn then specConn:Disconnect() specConn = nil end
        if v then
            specConn = RunService.Heartbeat:Connect(function()
                local tp = Players:FindFirstChild(selectedPlayerNameBH)
                if tp and tp.Character then
                    local h = tp.Character:FindFirstChildOfClass("Humanoid")
                    if h then workspace.CurrentCamera.CameraSubject = h end
                end
            end)
        else
            local chr = LocalPlayer.Character
            local h = chr and chr:FindFirstChildOfClass("Humanoid")
            if h then workspace.CurrentCamera.CameraSubject = h end
        end
    end })

    TabTroll:AddSection({ "Ações" })
    TabTroll:AddButton({ "Kill (Alvo)", function()
        local tp = Players:FindFirstChild(selectedPlayerNameBH)
        if tp then triggerCommand("KILL", tp.Name) end
    end })
    TabTroll:AddButton({ "Puxar Player", function()
        local tp = Players:FindFirstChild(selectedPlayerNameBH)
        if tp and tp.Character and tp.Character:FindFirstChild("HumanoidRootPart") then
            local h = getHRP()
            if h then tp.Character.HumanoidRootPart.CFrame = h.CFrame end
        end
    end })

    TabAudio:AddSection({ "Boombox & Músicas" })
    local customAudioId = ""
    TabAudio:AddTextBox({ Name = "ID da Música", Default = "", Placeholder = "1848354536",
        Callback = function(v) customAudioId = v end })
    TabAudio:AddButton({ "Tocar no Boombox", function()
        local id = tonumber(customAudioId)
        if not id then return end
        pcall(function()
            local br = ReplicatedStorage:FindFirstChild("RE") and ReplicatedStorage.RE:FindFirstChild("1BoomBox")
            if br then br:InvokeServer(id) end
        end)
    end })

    notify("Brookhaven", "Carregado com sucesso!")
end)
