local function getRealImageIdDelta(id)
    local clean = tostring(id):match("%d+")
    if not clean then return "rbxassetid://" .. tostring(id) end
    return "rbxassetid://" .. clean
end

local function fetchAssetAsModel(id)
    local s, objs = pcall(function() return game:GetObjects("rbxassetid://" .. tostring(id)) end)
    if s and objs and #objs > 0 then
        local m = Instance.new("Model")
        for _, o in ipairs(objs) do o.Parent = m end
        if not m.PrimaryPart then m.PrimaryPart = m:FindFirstChildWhichIsA("BasePart", true) end
        return m
    end
    return nil
end

    end)
end

local function sendNetCmd(cmd)
    pcall(function()
        if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
            local ch = TextChatService:FindFirstChild("TextChannels")
                and TextChatService.TextChannels:FindFirstChild("RBXGeneral")
            if ch then ch:SendAsync(cmd) end
        else
            local sayReq = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
                and ReplicatedStorage.DefaultChatSystemChatEvents:FindFirstChild("SayMessageRequest")
            if sayReq then sayReq:FireServer(cmd, "All") end
        end
    end)
end

-- ================== BLACKHOLE ==================
local function stopBlackhole(silent)
    if not BLACKHOLE.active then return end
    BLACKHOLE.active = false
    if BLACKHOLE.conn then pcall(function() BLACKHOLE.conn:Disconnect() end) end
    local n = 0
    for p, s in pairs(BLACKHOLE.saved) do
        if p and p.Parent then
            pcall(function()
                p.AssemblyLinearVelocity = s.vel
                p.AssemblyAngularVelocity = s.avel
            end)
            n += 1
        end
    end
    BLACKHOLE.saved = {}
    if BLACKHOLE.hole then pcall(function() BLACKHOLE.hole:Destroy() end) BLACKHOLE.hole = nil end
    if not silent then notify("Blackhole", n .. " objetos restaurados.") end
end

local function startBlackhole()
    if BLACKHOLE.active then stopBlackhole(true) end
    local hrp = getHRP()
    local pos = hrp and (hrp.Position + hrp.CFrame.LookVector * 140 + Vector3.new(0, 70, 0))
        or Vector3.new(0, 220, 0)
    BLACKHOLE.active = true
    local hole = Instance.new("Part")
    hole.Shape = Enum.PartType.Ball
    hole.Size = Vector3.new(28, 28, 28)
    hole.Position = pos
    hole.Anchored, hole.CanCollide = true, false
    hole.Material, hole.Color = Enum.Material.Neon, Color3.fromRGB(0, 0, 0)
    hole.Parent = workspace
    local glow = Instance.new("PointLight", hole)
    glow.Color = Color3.fromRGB(120, 30, 200)
    glow.Range, glow.Brightness = 220, 5
    BLACKHOLE.hole = hole
    BLACKHOLE.saved = {}
    local ex = { hole, LocalPlayer.Character }
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then table.insert(ex, p.Character) end
    end
    local params = OverlapParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = ex
    params.MaxParts = 900
    local found = workspace:GetPartBoundsInRadius(pos, 3500, params)
    local count = 0
    for _, p in ipairs(found) do
        if p:IsA("BasePart") and not p.Anchored then
            if not BLACKHOLE.saved[p] then
                BLACKHOLE.saved[p] = { vel = p.AssemblyLinearVelocity, avel = p.AssemblyAngularVelocity }
                count += 1
            end
        end
    end
    notify("Blackhole", count .. " objetos sugados!")
    BLACKHOLE.conn = RunService.Heartbeat:Connect(function(dt)
        if not BLACKHOLE.active or not hole.Parent then return end
        for p, s in pairs(BLACKHOLE.saved) do
            if p and p.Parent then
                local d = hole.Position - p.Position
                local dist = math.max(d.Magnitude, 4)
                p.AssemblyLinearVelocity = d.Unit * math.clamp(200000/(dist*dist), 40, 700)
            else
                BLACKHOLE.saved[p] = nil
            end
        end
    end)
end

-- ================== KILL LOOP ==================
local function setKillLoop(on)
    if KILLLOOP.conn then pcall(function() KILLLOOP.conn:Disconnect() end) KILLLOOP.conn = nil end
    if KILLLOOP.folder then pcall(function() KILLLOOP.folder:Destroy() end) KILLLOOP.folder = nil end
    if not on then notify("Kill Loop", "Desligado.") return end
    notify("Kill Loop", "VOCE entrou no loop de morte!")
    local fx = Instance.new("Folder", workspace)
    fx.Name = "F1L_KillLoopFX"
    KILLLOOP.folder = fx
    local blocks = {}
    for i = 1, 8 do
        local b = Instance.new("Part")
        b.Size = Vector3.new(3, 3, 3)
        b.Material = Enum.Material.Neon
        b.Color = Color3.fromRGB(255, 0, 0)
        b.Anchored, b.CanCollide = true, false
        b.Parent = fx
        blocks[i] = b
    end
    local angle = 0
    KILLLOOP.conn = RunService.Heartbeat:Connect(function(dt)
        local c = LocalPlayer.Character
        local hrp = c and c:FindFirstChild("HumanoidRootPart")
        local hum = c and c:FindFirstChildOfClass("Humanoid")
        if not hrp then return end
        angle += dt * 5
        for i, b in ipairs(blocks) do
            local a = angle + i * math.pi / 4
            b.CFrame = CFrame.new(hrp.Position + Vector3.new(
                math.cos(a) * 7, 2 + math.sin(angle*2+i)*1.5, math.sin(a) * 7))
        end
        if hum and hum.Health > 0 then hum.Health = 0 end
    end)
end

local function clearJailVisual(targetName)
    local st = JAILFX.built[targetName]
    if st then
        if st.conn then pcall(function() st.conn:Disconnect() end) end
        if st.model then pcall(function() st.model:Destroy() end) end
        JAILFX.built[targetName] = nil
    end
end

local function spawnJailVisual(targetName)
    clearJailVisual(targetName)
    local tp = Players:FindFirstChild(targetName)
    if not tp then return end
    local st = {}
    JAILFX.built[targetName] = st
    local function build(char)
        if st.model then pcall(function() st.model:Destroy() end) end
        local thrp = char and char:FindFirstChild("HumanoidRootPart")
        if not thrp then return end
        local cage = Instance.new("Model")
        cage.Name = "F1L_JailFX_" .. targetName
        local cf = thrp.CFrame
        for _, d in ipairs({
            { Vector3.new(7, 0.6, 7),   cf * CFrame.new(0, -3.2, 0) },
            { Vector3.new(7, 0.6, 7),   cf * CFrame.new(0,  3.6, 0) },
            { Vector3.new(0.6, 7, 7),   cf * CFrame.new( 3.2, 0, 0) },
            { Vector3.new(0.6, 7, 7),   cf * CFrame.new(-3.2, 0, 0) },
            { Vector3.new(7, 7, 0.6),   cf * CFrame.new(0, 0,  3.2) },
            { Vector3.new(7, 7, 0.6),   cf * CFrame.new(0, 0, -3.2) },
        }) do
            local p = Instance.new("Part", cage)
            p.Size, p.CFrame = d[1], d[2]
            p.Material, p.Color = Enum.Material.Neon, Color3.fromRGB(168, 85, 247)
            p.Transparency, p.Anchored, p.CanCollide = 0.35, true, false
        end
        cage.Parent = workspace
        st.model = cage
    end
    if tp.Character then task.spawn(build, tp.Character) end
    st.conn = tp.CharacterAdded:Connect(function(c)
        task.wait(1)
        if JAILFX.built[targetName] == st then build(c) end
    end)
end

local FLINGBUSY = false
local function doFlingTarget(targetName)
    local tp = Players:FindFirstChild(targetName)
    if not tp or tp == LocalPlayer or FLINGBUSY then return end
    local tr = tp.Character and tp.Character:FindFirstChild("HumanoidRootPart")
    local thum = tp.Character and tp.Character:FindFirstChildOfClass("Humanoid")
    if not tr then return end
    FLINGBUSY = true
    notify("Fling", "Lancando " .. tp.Name)
    local start = tick()
    while tick() - start < 3 and tr and tr.Parent and (not thum or thum.Health > 0) do
        applyNetworkBypass()
        pcall(function()
            tr.AssemblyAngularVelocity = Vector3.new(9e8, 9e8, 9e8)
            tr.AssemblyLinearVelocity = Vector3.new(math.random(-9e8, 9e8), 9e8, math.random(-9e8, 9e8))
        end)
        task.wait(0.05)
    end
    FLINGBUSY = false
    notify("Fling", tp.Name .. " foi lancado!")
end

-- ================== FLY ==================
local function toggleFlyWithWings()
    flyWingsActive = not flyWingsActive
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local torso = char and (char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso"))
    local head = char and char:FindFirstChild("Head")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not char or not hrp or not torso or not head then return end

    if flyWingsActive then
        local ok, w = pcall(function() return game:GetObjects("rbxassetid://" .. WINGS_ID) end)
        if ok and w and #w > 0 then
            wingsModel = w[1]
            if not wingsModel:IsA("Model") then
                local m = Instance.new("Model")
                for _, o in ipairs(w) do o.Parent = m end
                wingsModel = m
            end
            wingsModel.Parent = char
            for _, p in ipairs(wingsModel:GetDescendants()) do
                if p:IsA("BasePart") then p.Anchored, p.CanCollide = false, false end
            end
            local pr = wingsModel.PrimaryPart or wingsModel:FindFirstChildWhichIsA("BasePart", true)
            if pr then
                wingsModel.PrimaryPart = pr
                local wc = Instance.new("WeldConstraint")
                wc.Part0, wc.Part1, wc.Parent = torso, pr, pr
            end
        end
        local function loadHalo(id)
            local s, o = pcall(function() return game:GetObjects("rbxassetid://" .. id) end)
            if s and o and #o > 0 then
                local h = o[1]
                if not h:IsA("Model") then
                    local m = Instance.new("Model")
                    for _, x in ipairs(o) do x.Parent = m end
                    h = m
                end
                h.Parent = char
                for _, p in ipairs(h:GetDescendants()) do
                    if p:IsA("BasePart") then p.Anchored, p.CanCollide = false, false end
                end
                local pr = h.PrimaryPart or h:FindFirstChildWhichIsA("BasePart", true)
                if pr then
                    h.PrimaryPart = pr
                    local wc = Instance.new("WeldConstraint")
                    wc.Part0, wc.Part1, wc.Parent = head, pr, pr
                end
                return h
            end
        end
        halo1Model = loadHalo(HALO1_ID)
        halo2Model = loadHalo(HALO2_ID)
        local bv = Instance.new("BodyVelocity", hrp)
        bv.Name, bv.MaxForce, bv.Velocity = "FlyVelocity", Vector3.new(1e9,1e9,1e9), Vector3.zero
        local bg = Instance.new("BodyGyro", hrp)
        bg.Name, bg.MaxTorque, bg.CFrame = "FlyGyro", Vector3.new(1e9,1e9,1e9), hrp.CFrame
        if hum then hum.PlatformStand = true end
        local an = hum and hum:FindFirstChild("Animator")
        local ia = Instance.new("Animation") ia.AnimationId = FLY_IDLE
        local ma = Instance.new("Animation") ma.AnimationId = FLY_MOVE
        local it = an and an:LoadAnimation(ia)
        local mt = an and an:LoadAnimation(ma)
        if it then it:Play() end
        local lastM, cam = false, workspace.CurrentCamera
        local wt, ht = 0, 0
        flyConnections.RS = RunService.RenderStepped:Connect(function(dt)
            if not hrp or not hrp.Parent then return end
            local dir = Vector3.zero
            local ccf = cam.CFrame
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + ccf.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - ccf.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - ccf.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + ccf.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0,1,0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0,1,0) end
            if hum and hum.MoveDirection.Magnitude > 0 then dir = dir + hum.MoveDirection end
            if dir.Magnitude > 0 then
                bv.Velocity = dir.Unit * 55
                if not lastM then
                    lastM = true
                    if it then it:Stop() end
                    if mt and not mt.IsPlaying then mt:Play() end
                end
            else
                bv.Velocity = Vector3.zero
                if lastM then
                    lastM = false
                    if mt then mt:Stop() end
                    if it and not it.IsPlaying then it:Play() end
                end
            end
            bg.CFrame = ccf
            wt = wt + dt * 5
            if wingsModel and wingsModel.PrimaryPart then
                wingsModel:SetPrimaryPartCFrame(torso.CFrame * CFrame.new(0,-0.2,0.6) * CFrame.Angles(0, math.pi, math.sin(wt)*0.15))
            end
            ht = ht + dt
            local r = tick() * 3.5
            if halo1Model and halo1Model.PrimaryPart then
                halo1Model:SetPrimaryPartCFrame(head.CFrame * CFrame.new(0,1.3,0) * CFrame.Angles(0, r, 0))
            end
            if halo2Model and halo2Model.PrimaryPart then
                halo2Model:SetPrimaryPartCFrame(head.CFrame * CFrame.new(0,1.3,0) * CFrame.Angles(0, -r, 0))
            end
        end)
    else
        for _, c in pairs(flyConnections) do c:Disconnect() end
        flyConnections = {}
        if wingsModel then wingsModel:Destroy() wingsModel = nil end
        if halo1Model then halo1Model:Destroy() halo1Model = nil end
        if halo2Model then halo2Model:Destroy() halo2Model = nil end
        if hrp then
            if hrp:FindFirstChild("FlyVelocity") then hrp.FlyVelocity:Destroy() end
            if hrp:FindFirstChild("FlyGyro") then hrp.FlyGyro:Destroy() end
        end
        if hum then hum.PlatformStand = false end
    end
end

        pcall(function() car:ScaleTo(0.35) end)
        for _, p in ipairs(car:GetDescendants()) do
            if p:IsA("BasePart") then p.Anchored, p.CanCollide = true, false end
        end
        local sp = th.Position + th.CFrame.LookVector * 45
        local bcf, bs = car:GetBoundingBox()
        local bo = car:GetPivot().Position.Y - (bcf.Position.Y - bs.Y / 2)
        local tcf = CFrame.new(sp.X, sp.Y + bo, sp.Z) * CFrame.lookAt(sp, th.Position).Rotation
        car:PivotTo(tcf)
        local st, dur, hit = tick(), 10, false
        while tick() - st < dur and car and car.Parent do
            if tp and tp.Character then
                local cth = tp.Character:FindFirstChild("HumanoidRootPart")
                local ch = tp.Character:FindFirstChildOfClass("Humanoid")
                if cth then
                    local ccf = car:GetPivot()
                    local tpos = Vector3.new(cth.Position.X, cth.Position.Y + bo, cth.Position.Z)
                    local d = tpos - ccf.Position
                    if d.Magnitude > 2 then car:PivotTo(CFrame.new(ccf.Position + d.Unit * 3.75, tpos)) end
                    if (ccf.Position - cth.Position).Magnitude <= 7 and not hit then
                        hit = true
                        if ch then ch.Health = 0 end
                        Instance.new("Explosion", workspace).Position = cth.Position
                    end
                end
            else break end
            task.wait(0.03)
        end
        task.wait(0.4)
        if car then car:Destroy() end
    end)
end

local function spawnLifeform(spawnPos, folder, targetPlayerName)
    if lifeformState.model and lifeformState.model.Parent then
        lifeformState.model:Destroy()
    end
    local targetFolder = folder or workspace:FindFirstChild("Backrooms_Level0") or workspace
    local model = Instance.new("Model", targetFolder)
    model.Name = "TheLifeform"
    local body = Instance.new("Part", model)
    body.Name = "Body"
    body.Size = Vector3.new(2, 5, 1)
    body.CFrame = CFrame.new(spawnPos)
    body.Anchored, body.CanCollide, body.Transparency = true, false, 1
    body.CastShadow = false
    model.PrimaryPart = body

    local bb = Instance.new("BillboardGui", body)
    bb.Size = UDim2.new(0, 170, 0, 240)
    bb.StudsOffset = Vector3.new(0, 2.5, 0)
    bb.LightInfluence = 0
    bb.MaxDistance = 250
    local img = Instance.new("ImageLabel", bb)
    img.Size = UDim2.new(1,0,1,0)
    img.BackgroundTransparency = 1
    img.Image = LIFEFORM_IMG
    img.ScaleType = Enum.ScaleType.Fit

    local snd = Instance.new("Sound", body)
    snd.SoundId, snd.Looped, snd.Volume = LIFEFORM_SOUND, true, 4
    snd.RollOffMaxDistance = 200
    snd.RollOffMode = Enum.RollOffMode.InverseTapered

    local light = Instance.new("PointLight", body)
    light.Color, light.Brightness, light.Range = Color3.fromRGB(180, 0, 0), 3, 30

    lifeformState.model = model
    lifeformState.active = true

    task.spawn(function()
        local path = PathfindingService:CreatePath({
            AgentRadius = 2, AgentHeight = 5, AgentCanJump = false, AgentMaxSlope = 45,
        })
        local waypoints, wpi, cooldown = {}, 1, 0
        local SPEED, DETECT_RANGE = 22, 500
        while lifeformState.active and model.Parent do
            local targetPlr = Players:FindFirstChild(targetPlayerName) or LocalPlayer
            local chr = targetPlr and targetPlr.Character
            local th = chr and chr:FindFirstChild("HumanoidRootPart")
            local thum = chr and chr:FindFirstChildOfClass("Humanoid")
            if th and thum and thum.Health > 0 then
                local myPos = body.Position
                local targetPos = th.Position
                local dist = (targetPos - myPos).Magnitude
                if dist <= DETECT_RANGE then
                    if not snd.IsPlaying then pcall(function() snd:Play() end) end
                    if tick() - cooldown > 0.4 then
                        cooldown = tick()
                        pcall(function() path:ComputeAsync(myPos, targetPos) end)
                        if path.Status == Enum.PathStatus.Success then
                            waypoints = path:GetWaypoints()
                            wpi = 2
                        end
                    end
                    local moveTarget = targetPos
                    if wpi <= #waypoints then
                        local wp = waypoints[wpi]
                        moveTarget = wp.Position
                        if (Vector3.new(myPos.X,0,myPos.Z) - Vector3.new(wp.Position.X,0,wp.Position.Z)).Magnitude < 3 then
                            wpi = wpi + 1
                        end
                    end
                    local dir = moveTarget - myPos
                    dir = Vector3.new(dir.X, 0, dir.Z)
                    if dir.Magnitude > 0.5 then
                        dir = dir.Unit
                        local newPos = myPos + dir * SPEED * 0.05
                        body.CFrame = CFrame.lookAt(
                            Vector3.new(newPos.X, myPos.Y, newPos.Z),
                            Vector3.new(targetPos.X, myPos.Y, targetPos.Z))
                    end
                    if dist < 4 then thum.Health = 0 task.wait(2) end
                else
                    if snd.IsPlaying then pcall(function() snd:Stop() end) end
                end
            end
            task.wait(0.05)
        end
    end)
    notify("The Lifeform", "Ele acordou perseguindo " .. targetPlayerName .. ".")
    return model
end

-- ================== POOLROOMS ==================
do
    local CH, OFFS, PANEL_OFFS = 64, { -24, -8, 8, 24 }, { -32, -16, 0, 16 }

    local RADIUS, TILE, LIGHT_COUNT, SHADOW_COUNT = 2, 4, 36, 10
    local WALL_TEX = "rbxassetid://122349893272800"

    local C_TILE  = Color3.fromRGB(246, 248, 248)
    local C_FLOOR = Color3.fromRGB(222, 238, 236)
    local C_WATER = Color3.fromRGB(70, 205, 198)
    local C_WET   = Color3.fromRGB(60, 190, 185)
    local C_TRIM  = Color3.fromRGB(170, 178, 180)
    local C_CEIL  = Color3.fromRGB(244, 246, 248)
    local C_FRAME = Color3.fromRGB(150, 155, 160)
    local C_PANEL = Color3.fromRGB(232, 244, 255)
    local C_LIGHT = Color3.fromRGB(214, 232, 255)

    local FACES_FB    = { Enum.NormalId.Front, Enum.NormalId.Back }
    local FACES_LR    = { Enum.NormalId.Left, Enum.NormalId.Right }
    local FACES_TOP   = { Enum.NormalId.Top }
    local FACES_SIDES = { Enum.NormalId.Front, Enum.NormalId.Back, Enum.NormalId.Left, Enum.NormalId.Right }

    local imageCache = {}
    local function resolveImage(id)
        local clean = tostring(id):match("%d+")
        if not clean then return tostring(id) end
        if imageCache[clean] then return imageCache[clean] end
        local fallback = "rbxassetid://" .. clean
        local result = fallback
        local ok, objs = pcall(function() return game:GetObjects(fallback) end)
        if ok and objs and objs[1] then
            local o = objs[1]
            local tex
            if o:IsA("Decal") or o:IsA("Texture") then tex = o.Texture end
            if tex and tex ~= "" then result = tex end
            pcall(function() o:Destroy() end)
        end
        imageCache[clean] = result
        return result
    end

    local function mkPart(parent, size, cf, color, mat, collide)
        local p = Instance.new("Part")
        p.Anchored = true
        p.Size, p.CFrame, p.Color = size, cf, color
        p.Material = mat or Enum.Material.SmoothPlastic
        p.CanCollide = collide and true or false
        p.CanQuery = collide and true or false
        p.CanTouch = false
        p.Parent = parent
        return p
    end

    local function slab(S, parent, size, cf, faces, color, collide)
        local p = mkPart(parent, size, cf, color or C_TILE, Enum.Material.SmoothPlastic, collide ~= false)
        p.Reflectance = 0.12
        for _, f in ipairs(faces or FACES_FB) do
            local t = Instance.new("Texture")
            t.Texture, t.Face = S.tex, f
            t.StudsPerTileU, t.StudsPerTileV = TILE, TILE
            t.Parent = p
        end
        S.count += 1
        if S.count % 70 == 0 then task.wait() end
        return p
    end

    local function buildChunk(S, cx, cz)
        if POOL.token ~= S.token then return end
        local key = cx .. "," .. cz
        if S.chunks[key] then return end
        local folder = Instance.new("Folder")
        folder.Name = "Chunk_" .. key
        folder.Parent = S.folder
        local chunk = { folder = folder, cx = cx, cz = cz, panels = {} }
        S.chunks[key] = chunk

        local rng = Random.new(cx * 7349 + cz * 9157 + 11)
        local wx, wz = cx * CH, cz * CH
        local ceilBottom = POOL_Y + WALL_H

        pcall(function()
            slab(S, folder, Vector3.new(CH, 2, CH), CFrame.new(wx, POOL_Y - 1, wz), FACES_TOP, C_FLOOR)
            local w = mkPart(folder, Vector3.new(CH, WATER_H, CH),
                CFrame.new(wx, POOL_Y + WATER_H/2, wz), C_WATER, Enum.Material.Glass, false)
            w.Transparency, w.Reflectance, w.CastShadow = 0.55, 0.22, false
            local ceil = mkPart(folder, Vector3.new(CH, 2, CH),
                CFrame.new(wx, ceilBottom + 1, wz), C_CEIL, Enum.Material.SmoothPlastic, true)
            ceil.Reflectance = 0.04

            for _, dx in ipairs(OFFS) do
                for _, dz in ipairs(OFFS) do
                    local x, z = wx + dx, wz + dz
                    slab(S, folder, Vector3.new(COL_W, WALL_H, COL_W),
                        CFrame.new(x, POOL_Y + WALL_H/2, z), FACES_SIDES, C_TILE)
                    local b = mkPart(folder, Vector3.new(COL_W+0.14, 3, COL_W+0.14),

                    b.Transparency, b.Reflectance, b.CastShadow = 0.6, 0.1, false
                    mkPart(folder, Vector3.new(COL_W+0.9, 0.8, COL_W+0.9),
                        CFrame.new(x, ceilBottom - 0.4, z), C_TRIM, Enum.Material.SmoothPlastic, false)
                end
            end

            for _, u in ipairs(PANEL_OFFS) do
                for _, v in ipairs(PANEL_OFFS) do
                    local x, z = wx + u, wz + v
                    local frame = mkPart(folder, Vector3.new(10.4, 0.5, 4.4),
                        CFrame.new(x, ceilBottom - 0.25, z), C_FRAME, Enum.Material.Metal, false)
                    frame.CastShadow = false
                    local pan = mkPart(folder, Vector3.new(9.6, 0.3, 3.6),
                        CFrame.new(x, ceilBottom - 0.55, z), C_PANEL, Enum.Material.Neon, false)
                    pan.CastShadow = false
                    table.insert(chunk.panels, { part = pan, light = nil, d = 0 })
                end
            end
        end)
    end

    local function unloadChunk(S, key)
        local ch = S.chunks[key]
        if not ch then return end
        S.chunks[key] = nil
        pcall(function() ch.folder:Destroy() end)
    end

    local function streamStep(S, pcx, pcz, radius)
        for ring = 0, radius do
            for dx = -ring, ring do
                for dz = -ring, ring do
                    if POOL.token ~= S.token then return end
                    if math.max(math.abs(dx), math.abs(dz)) == ring then
                        buildChunk(S, pcx + dx, pcz + dz)
                    end
                end
            end
        end
        for key, ch in pairs(S.chunks) do
            if math.max(math.abs(ch.cx - pcx), math.abs(ch.cz - pcz)) > RADIUS + 1 then
                unloadChunk(S, key)
            end
        end
    end

    local function updateLights(S, pos)
        local list = {}
        for _, ch in pairs(S.chunks) do
            for _, pn in ipairs(ch.panels) do
                if pn.part.Parent then
                    pn.d = (pn.part.Position - pos).Magnitude
                    list[#list + 1] = pn
                end
            end
        end
        table.sort(list, function(a, b) return a.d < b.d end)
        for i, pn in ipairs(list) do
            if i <= LIGHT_COUNT then
                if not (pn.light and pn.light.Parent) then
                    local l = Instance.new("SurfaceLight")
                    l.Face = Enum.NormalId.Bottom
                    l.Angle, l.Range, l.Brightness = 170, 58, 1.7
                    l.Color, l.Shadows = C_LIGHT, false
                    l.Parent = pn.part
                    pn.light = l
                end
                pn.light.Shadows = i <= SHADOW_COUNT
            elseif pn.light then
                pn.light:Destroy()
                pn.light = nil
            end
        end
    end

    local LIGHT_PROPS = {
        "Ambient", "OutdoorAmbient", "Brightness", "ColorShift_Top", "ColorShift_Bottom",
        "ExposureCompensation", "ClockTime", "FogColor", "FogStart", "FogEnd", "GlobalShadows",
    }
    local function envSave()
        local s = {}
        for _, k in ipairs(LIGHT_PROPS) do pcall(function() s[k] = Lighting[k] end) end
        return s
    end
    local function envApply()
        local function set(k, v) pcall(function() Lighting[k] = v end) end
        set("ClockTime", 14); set("Brightness", 2.2)
        set("Ambient", Color3.fromRGB(125, 148, 156))
        set("OutdoorAmbient", Color3.fromRGB(140, 160, 168))
        set("FogColor", Color3.fromRGB(205, 228, 236))
        set("FogStart", 70); set("FogEnd", 250)
        pcall(function()
            for _, c in ipairs(Lighting:GetChildren()) do
                if c:IsA("Atmosphere") then
                    table.insert(POOL.hidden, c)
                    c.Parent = nil
                end
            end
        end)
        pcall(function()
            local cc = Instance.new("ColorCorrectionEffect")
            cc.Name = "Pool_CC"
            cc.TintColor = Color3.fromRGB(235, 248, 255)
            cc.Saturation, cc.Contrast, cc.Brightness = 0.1, 0.08, 0.02
            cc.Parent = Lighting
            local at = Instance.new("Atmosphere")
            at.Name = "Pool_Atmosphere"
            at.Density, at.Offset = 0.22, 0
            at.Color, at.Decay = Color3.fromRGB(210, 230, 238), Color3.fromRGB(150, 180, 190)
            at.Parent = Lighting
        end)
    end
    local function envRestore()
        local s = POOL.saved
        if not s then return end
        POOL.saved = nil
        for _, k in ipairs(LIGHT_PROPS) do
            if s[k] ~= nil then pcall(function() Lighting[k] = s[k] end) end
        end
    end

    function POOL.clear(silent)
        POOL.token += 1
        local S = POOL.S
        POOL.S = nil
        if S then
            if S.conn then pcall(function() S.conn:Disconnect() end) end
            if S.folder then pcall(function() S.folder:Destroy() end) end
        end
        for _, n in ipairs({ "Pool_CC", "Pool_Atmosphere" }) do
            local o = Lighting:FindFirstChild(n)
            if o then o:Destroy() end
        end
        for _, a in ipairs(POOL.hidden) do pcall(function() a.Parent = Lighting end) end
        POOL.hidden = {}
        envRestore()
        if S and not silent then notify("Poolrooms", "Poolrooms removidas.") end
        return true
    end

    function POOL.spawn(partnerName)
        notify("Poolrooms", "Gerando Poolrooms...")
        POOL.clear(true)

        folder.Parent = workspace
        S.folder = folder

        pcall(function() S.tex = resolveImage(WALL_TEX) end)
        if POOL.token ~= myToken then return end

        POOL.saved = envSave()
        envApply()

        streamStep(S, 0, 0, 1)
        if POOL.token ~= myToken then return end

        local spawnCF = CFrame.new(16, POOL_Y + 6, 16)
        task.spawn(function()
            task.wait(0.3)
            local chr = LocalPlayer.Character
            local hrp = chr and chr:FindFirstChild("HumanoidRootPart")
            if hrp then
                hrp.Anchored = true
                hrp.CFrame = spawnCF
                task.wait(0.3)
                hrp.Anchored = false
            end
            if partnerName and partnerName ~= "" then
                local partner = Players:FindFirstChild(partnerName)
                if partner and partner.Character and partner.Character:FindFirstChild("HumanoidRootPart") then
                    partner.Character.HumanoidRootPart.CFrame = spawnCF + Vector3.new(3, 0, 3)
                end
            end
        end)

        S.conn = LocalPlayer.CharacterAdded:Connect(function(c)
            local h = c:WaitForChild("HumanoidRootPart", 10)
            if h and POOL.token == myToken then
                task.wait(0.5)
                pcall(streamStep, S, 0, 0, 1)
                if POOL.token ~= myToken then return end
                h.AssemblyLinearVelocity = Vector3.zero
                h.CFrame = spawnCF
            end
        end)

        task.spawn(function()
            while POOL.token == myToken do
                local c = LocalPlayer.Character
                local h = c and c:FindFirstChild("HumanoidRootPart")
                if h then
                    local y = h.Position.Y
                    local inPool = y > POOL_Y - 40 and y < POOL_Y + 300
                    local pcx, pcz = 0, 0
                    if inPool then
                        pcx = math.floor((h.Position.X + CH/2) / CH)
                        pcz = math.floor((h.Position.Z + CH/2) / CH)
                    end
                    pcall(streamStep, S, pcx, pcz, RADIUS)
                    pcall(updateLights, S, inPool and h.Position or spawnCF.Position)
                    if y < POOL_Y - 20 and y > POOL_Y - 600 then
                        pcall(streamStep, S, 0, 0, 1)
                        h.AssemblyLinearVelocity = Vector3.zero
                        h.CFrame = spawnCF
                    end
                end
                task.wait(0.35)
            end
        end)

        notify("Poolrooms", "Pronto!")
        return true
    end
end

do

    local function fmt(n)
        for _, u in ipairs({ { 1e12, "T" }, { 1e9, "B" }, { 1e6, "M" }, { 1e3, "K" } }) do
            if n >= u[1] then
                local s = string.format("%.2f", n / u[1])
                s = (s:gsub("(%..-)0+$", "%1")):gsub("%.$", "")
                return s .. u[2]
            end
        end
        return tostring(math.floor(n))
    end

    local function pickDef(maxTier)
        local pool, total = {}, 0
        for i, d in ipairs(BRAINROTS) do
            if not maxTier or i <= maxTier then
                table.insert(pool, d)
                total += d.w
            end
        end
        local r = math.random() * total
        for _, d in ipairs(pool) do
            r -= d.w
            if r <= 0 then return d end
        end
        return pool[1]
    end

    local function mk(parent, size, cf, color, mat, collide)
        local p = Instance.new("Part")
        p.Anchored = true
        p.Size, p.CFrame, p.Color = size, cf, color
        p.Material = mat or Enum.Material.SmoothPlastic
        p.CanCollide = collide ~= false
        p.Parent = parent
        BRAIN.count += 1
        if BRAIN.count % 60 == 0 then task.wait() end
        return p
    end

    local function fetchObjects(id)
        for _ = 1, 2 do
            for _, url in ipairs({ "rbxassetid://"..id, "http://www.roblox.com/asset/?id="..id }) do
                local ok, objs = pcall(function() return game:GetObjects(url) end)
                if ok and objs and #objs > 0 then return objs end
            end
            task.wait(0.4)
        end
        return nil
    end

    local function loadTemplate(def)
        local c = BRAIN.cache[def.asset]
        if c ~= nil then return c or nil end
        local objs = fetchObjects(def.asset)
        local tpl
        if objs then
            local m = Instance.new("Model")
            for _, o in ipairs(objs) do o.Parent = m end
            for _, d in ipairs(m:GetDescendants()) do
                if d:IsA("LuaSourceContainer") or d:IsA("Humanoid") then pcall(function() d:Destroy() end) end
            end
            for _, d in ipairs(m:GetDescendants()) do
                if d:IsA("BasePart") then tpl = m break end
            end
        end
        BRAIN.cache[def.asset] = tpl or false
        return tpl
    end

    local function makeBrainrot(def, parent)
        local tpl = loadTemplate(def)
        local m = Instance.new("Model")
        m.Name = def.name
        local root = Instance.new("Part")
        root.Name, root.Size = "Root", Vector3.new(1,1,1)
        root.Transparency, root.Anchored, root.CanCollide = 1, true, false
        root.Parent = m
        m.PrimaryPart = root

        local height
        if tpl then
            local clone = tpl:Clone()
            for _, d in ipairs(clone:GetDescendants()) do
                if d:IsA("BasePart") then d.Anchored, d.CanCollide = true, false end
            end
            clone.Parent = m
            local cf, size = clone:GetBoundingBox()
            root.CFrame = CFrame.new(cf.Position.X, cf.Position.Y - size.Y/2, cf.Position.Z)
            height = size.Y
        else
            local function part(sz, off, col)
                local p = Instance.new("Part")
                p.Anchored, p.CanCollide = true, false
                p.Size, p.Color = sz, col
                p.CFrame = CFrame.new(off)
                p.Parent = m
                return p
            end
            part(Vector3.new(3.2, 3.6, 2.2), Vector3.new(0, 3.2, 0), def.col)
            part(Vector3.new(3.4, 3, 3), Vector3.new(0, 5.9, 0), def.col2)
            root.CFrame = CFrame.new(0, 0, 0)
            height = 7.5
        end

        local anchor = Instance.new("Attachment")
        anchor.Name, anchor.Position = "PromptAnchor", Vector3.new(0, height/2, 0)
        anchor.Parent = root

        local bb = Instance.new("BillboardGui")
        bb.Size = UDim2.new(0, 170, 0, 76)
        bb.StudsOffset = Vector3.new(0, height + 2.5, 0)
        bb.MaxDistance, bb.Adornee = 160, root
        bb.Parent = root
        local function lab(txt, color, y)
            local l = Instance.new("TextLabel")
            l.BackgroundTransparency = 1
            l.Size, l.Position = UDim2.new(1, 0, 0.25, 0), UDim2.new(0, 0, y, 0)
            l.Text, l.TextColor3 = txt, color
            l.TextStrokeTransparency = 0
            l.Font, l.TextScaled = Enum.Font.GothamBold, true
            l.Parent = bb
        end
        lab(def.name, Color3.new(1,1,1), 0)
        lab(def.rarity, def.rc, 0.25)
        lab("$" .. fmt(def.inc) .. "/s", Color3.fromRGB(90, 255, 110), 0.5)
        lab("$" .. fmt(def.price), Color3.fromRGB(255, 225, 80), 0.75)
        m.Parent = parent
        return m
    end

    local function setPrompt(model, action, hold, cb)
        local root = model.PrimaryPart
        local anchor = root:FindFirstChild("PromptAnchor") or root
        local old = anchor:FindFirstChildOfClass("ProximityPrompt")
        if old then old:Destroy() end
        local pp = Instance.new("ProximityPrompt")
        pp.ActionText, pp.ObjectText = action, model.Name
        pp.HoldDuration, pp.MaxActivationDistance = hold, 14
        pp.RequiresLineOfSight = false
        pp.Parent = anchor
        pp.Triggered:Connect(cb)
        return pp
    end

    local function setMoney(S)
        if S.hudMoney then S.hudMoney.Text = "  $" .. fmt(S.money) end
        if S.hudInc then S.hudInc.Text = "  +$" .. fmt(S.income) .. "/s" end

    local function updateCollect(S)
        if S.collectLabel then S.collectLabel.Text = "$" .. fmt(S.pending) end
    end

    local function freeSlot(slots)
        for _, s in ipairs(slots) do
            if not s.model then return s end
        end
    end

    local function placeModel(model, slot)
        local p = slot.pos
        model:PivotTo(CFrame.lookAt(p, p + slot.look))
    end

    local function claimMine(S, model, def, slot)
        slot.model, slot.def = model, def
        placeModel(model, slot)
        S.income += def.inc
        setMoney(S)
        setPrompt(model, "Vender", 0.8, function()
            if slot.model ~= model then return end
            slot.model, slot.def = nil, nil
            S.money += math.floor(def.price / 2)
            S.income -= def.inc
            setMoney(S)
            model:Destroy()
        end)
    end

    local function putNpc(S, base, def, slot)
        local model = makeBrainrot(def, S.folder)
        slot.model, slot.def = model, def
        placeModel(model, slot)
        setPrompt(model, "Roubar", 1.4, function()
            if slot.model ~= model then return end
            local mine = freeSlot(S.mySlots)
            if not mine then notify("Brainrot", "Sua base esta cheia!") return end
            slot.model, slot.def = nil, nil
            claimMine(S, model, def, mine)
            notify("Brainrot", "Voce roubou " .. def.name .. "!")
        end)
    end

    local function buildBase(f, cx, cz, side, label, mine)
        local ang = (side < 0) and math.pi/2 or -math.pi/2
        local F = CFrame.new(cx, GY, cz) * CFrame.Angles(0, ang, 0)
        local function P(size, lx, ly, lz, color, mat, collide)
            return mk(f, size, F * CFrame.new(lx, ly, lz), color, mat, collide)
        end
        local halfW, halfD, H = BASE_W/2, BASE_D/2, 18

        P(Vector3.new(BASE_W + 4, 1, BASE_D + 4), 0, 0.5, 0, C_GRAY, Enum.Material.Plastic)
        P(Vector3.new(BASE_W - 6, 0.4, BASE_D - 6), 0, 1.2, 0, C_FLOOR, Enum.Material.SmoothPlastic)
        for _, sx in ipairs({-1,1}) do
            for _, sz in ipairs({-1,1}) do
                P(Vector3.new(5, H, 5), sx*(halfW-2.5), 1+H/2, sz*(halfD-2.5), C_GRAY, Enum.Material.Plastic)
            end
        end
        P(Vector3.new(BASE_W + 4, 2, BASE_D + 4), 0, 1+H+1, 0, C_GRAY, Enum.Material.Plastic)

        local cz0 = halfD + 15
        P(Vector3.new(16, 0.2, 16), 0, 0.1, cz0, C_WHITE, Enum.Material.SmoothPlastic, false)
        local pad = P(Vector3.new(14, 0.2, 14), 0, 0.3, cz0, Color3.fromRGB(40, 220, 70), Enum.Material.SmoothPlastic, false)

        local slots = {}
        local look = F:VectorToWorldSpace(Vector3.new(0, 0, 1))
        for _, lz in ipairs({ 18, 6, -6, -18 }) do
            for _, sx in ipairs({-1, 1}) do
                P(Vector3.new(8, 0.3, 8), sx*12, 1.55, lz, Color3.fromRGB(46, 214, 76), Enum.Material.SmoothPlastic, false)
                table.insert(slots, { pos = (F * CFrame.new(sx*12, 1.7, lz)).Position, look = look, model = nil })
            end
        end

        local result = { slots = slots, label = label }
        if mine then
            local gui = Instance.new("BillboardGui")
            gui.Size, gui.StudsOffset, gui.MaxDistance = UDim2.new(0, 220, 0, 50), Vector3.new(0, 5, 0), 200
            gui.Adornee = pad
            gui.Parent = pad
            local lbl = Instance.new("TextLabel")
            lbl.BackgroundTransparency = 1
            lbl.Size, lbl.Text = UDim2.new(1,0,1,0), "$0"
            lbl.TextScaled, lbl.Font = true, Enum.Font.GothamBlack
            lbl.TextColor3, lbl.TextStrokeTransparency = Color3.fromRGB(255,225,80), 0
            lbl.Parent = gui
            result.collectLabel = lbl
            result.collectPos = pad.Position
        end
        return result
    end

    function BRAIN.clear(silent)
        BRAIN.token += 1
        local S = BRAIN.S
        BRAIN.S = nil
        if S then
            if S.conn then pcall(function() S.conn:Disconnect() end) end
            if S.charConn then pcall(function() S.charConn:Disconnect() end) end
            if S.folder then pcall(function() S.folder:Destroy() end) end
            if S.gui then pcall(function() S.gui:Destroy() end) end
        end
        for _, n in ipairs({ "BR_Sky", "BR_CC" }) do
            local o = Lighting:FindFirstChild(n)
            if o then o:Destroy() end
        end
        for _, a in ipairs(BRAIN.hidden) do pcall(function() a.Parent = Lighting end) end
        BRAIN.hidden = {}
        local sv = BRAIN.saved
        BRAIN.saved = nil
        if sv then
            for k, v in pairs(sv) do pcall(function() Lighting[k] = v end) end
        end
        if S and not silent then notify("Brainrot", "Mundo removido.") end
        return true
    end

    function BRAIN.spawn()
        notify("Brainrot", "Gerando o mundo...")
        BRAIN.clear(true)
        BRAIN.count = 0
        local myToken = BRAIN.token

        local S = { money = START_MONEY, pending = 0, income = 0, walkers = {}, mySlots = {}, npcBases = {} }
        BRAIN.S = S
        local f = Instance.new("Folder")
        f.Name = "RoubeUmBrainrot_World"
        f.Parent = workspace
        S.folder = f

        BRAIN.saved = {}
        for _, k in ipairs(LIGHT_PROPS) do pcall(function() BRAIN.saved[k] = Lighting[k] end) end
        local function set(k, v) pcall(function() Lighting[k] = v end) end
        set("ClockTime", 14); set("Brightness", 2.4)
        set("Ambient", Color3.fromRGB(150,150,150))
        set("OutdoorAmbient", Color3.fromRGB(160,160,160))
        set("FogStart", 0); set("FogEnd", 100000)
        pcall(function()
            for _, c in ipairs(Lighting:GetChildren()) do
                if c:IsA("Atmosphere") or c:IsA("Sky") then
                    table.insert(BRAIN.hidden, c)
                    c.Parent = nil
                end
            end
        end)

        -- chão
        mk(f, Vector3.new(W, 4, L), CFrame.new(0, GY - 2, 0), C_GRASS, Enum.Material.Plastic)
        mk(f, Vector3.new(CARPET_W, 0.4, L), CFrame.new(0, GY + 0.2, 0), C_CARPET, Enum.Material.Plastic)
        -- paredes externas
        local EW = L + WALL_T * 2
        mk(f, Vector3.new(WALL_T, 8, EW), CFrame.new(-(W/2+WALL_T/2), GY+4, 0), C_STONE, Enum.Material.Slate)
        mk(f, Vector3.new(WALL_T, 8, EW), CFrame.new( (W/2+WALL_T/2), GY+4, 0), C_STONE, Enum.Material.Slate)
        for _, zs in ipairs({-1,1}) do
            local cz = zs * (L/2 + WALL_T/2)
            local seg = (W - GATE) / 2
            mk(f, Vector3.new(seg, 8, WALL_T), CFrame.new(-(GATE/2 + seg/2), GY+4, cz), C_STONE, Enum.Material.Slate)
            mk(f, Vector3.new(seg, 8, WALL_T), CFrame.new( (GATE/2 + seg/2), GY+4, cz), C_STONE, Enum.Material.Slate)
            mk(f, Vector3.new(GATE, 8, WALL_T), CFrame.new(0, GY + 8 + 8, cz), C_STONE, Enum.Material.Slate)
        end
        if BRAIN.token ~= myToken then return end

        -- bases
        local ni, idx = 0, 0
        for _, side in ipairs({-1, 1}) do
            for _, bz in ipairs(BASE_Z) do
                idx += 1
                local label
                if idx == 1 then label = LocalPlayer.DisplayName
                else ni += 1 label = NPC_NAMES[ni] or ("Player" .. ni) end
                local base = buildBase(f, side * BASE_X, bz, side, label, idx == 1)
                if idx == 1 then
                    S.mySlots = base.slots
                    S.collectLabel = base.collectLabel
                    S.collectPos = base.collectPos
                else
                    table.insert(S.npcBases, base)
                end
            end
        end

        makeHud(S)

        -- NPC brainrots
        for _, base in ipairs(S.npcBases) do
            for i = 1, math.random(2, 3) do
                local slot = freeSlot(base.slots)
                if slot then putNpc(S, base, pickDef(2), slot) end
            end
        end

        local startPos = Vector3.new(0, GY + 4, L/2 - 20)
        local startCF = CFrame.lookAt(startPos, Vector3.new(0, GY + 4, 0))
        task.spawn(function()
            task.wait(0.5)
            if BRAIN.token ~= myToken then return end
            local chr = LocalPlayer.Character
            local hrp = chr and chr:FindFirstChild("HumanoidRootPart")
            if hrp then
                hrp.Anchored = true
                hrp.CFrame = startCF
                task.wait(0.3)
                hrp.Anchored = false
            end
        end)

        S.charConn = LocalPlayer.CharacterAdded:Connect(function(c)
            local h = c:WaitForChild("HumanoidRootPart", 10)
            if h and BRAIN.token == myToken then
                task.wait(0.5)
                h.AssemblyLinearVelocity = Vector3.zero
                h.CFrame = startCF
            end
        end)

        S.conn = RunService.Heartbeat:Connect(function(dt)
            local pc = LocalPlayer.Character
            local ph = pc and pc:FindFirstChild("HumanoidRootPart")
            if ph and S.collectPos and S.pending > 0 then
                local d = ph.Position - S.collectPos
                if math.abs(d.X) < 8 and math.abs(d.Z) < 8 and math.abs(d.Y) < 8 then
                    S.money += S.pending
                    notify("Brainrot", "Coletou $" .. fmt(S.pending))
                    S.pending = 0
                    updateCollect(S)
                    setMoney(S)
                end
            end
        end)

        task.spawn(function()
            while BRAIN.token == myToken do
                task.wait(1)
                S.pending += S.income
                updateCollect(S)
            end
        end)

        notify("Brainrot", "Pronto! Segure E nos brainrots. Fique na ZONA DE COLETA.")
        return true
    end
end

local function roubeUmBrainrot(modo)
    modo = modo and string.upper(modo) or (BRAIN.S and "OFF" or "ON")
    if modo == "OFF" then BRAIN.clear() else BRAIN.spawn() end
end

-- ================== EXECUTOR LOCAL ==================
local function executeActionLocally(action, senderName, extraArg)
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")

    if action == "KILL" and hum then hum.Health = 0
    elseif action == "KICK" then LocalPlayer:Kick("Expulso pelo Fantasma1l Hub.")
    elseif action == "FREEZE" and hrp and hum then hrp.Anchored, hum.WalkSpeed, hum.JumpPower = true, 0, 0
    elseif action == "UNFREEZE" and hrp and hum then hrp.Anchored, hum.WalkSpeed, hum.JumpPower = false, 16, 50
    elseif action == "CRASH" then while true do end
    elseif action == "EXPLODE" and hrp then
        Instance.new("Explosion", workspace).Position = hrp.Position
        if hum then hum.Health = 0 end
    elseif action == "ICE" and hrp then
        if workspace:FindFirstChild("Ice_" .. LocalPlayer.Name) then workspace["Ice_" .. LocalPlayer.Name]:Destroy() end
        local ip = Instance.new("Part", workspace)
        ip.Name, ip.Size, ip.CFrame = "Ice_" .. LocalPlayer.Name, Vector3.new(6,8,6), hrp.CFrame
        ip.Anchored, ip.CanCollide = true, false
        ip.Color, ip.Material, ip.Transparency = Color3.fromRGB(0,170,255), Enum.Material.Glass, 0.4
        if hum then hum.WalkSpeed = 0 end
        hrp.Anchored = true
    elseif action == "UNICE" and hrp then
        if workspace:FindFirstChild("Ice_" .. LocalPlayer.Name) then workspace["Ice_" .. LocalPlayer.Name]:Destroy() end
        if hum then hum.WalkSpeed = 16 end
        hrp.Anchored = false
    elseif action == "TP_TO_POS" and hrp and senderName then
        local sp = Players:FindFirstChild(senderName)
        if sp and sp.Character and sp.Character:FindFirstChild("HumanoidRootPart") then
            hrp.CFrame = sp.Character.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)
        end
    elseif action == "TP_ME" and hrp and senderName then
        local sp = Players:FindFirstChild(senderName)
        if sp and sp.Character and sp.Character:FindFirstChild("HumanoidRootPart") then
            sp.Character.HumanoidRootPart.CFrame = hrp.CFrame * CFrame.new(0, 0, -3)
        end
    elseif action == "VISIBLE" and char then
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") then p.Transparency = 0 p.LocalTransparencyModifier = 0 end
        end
    elseif action == "CHAT_AS" and extraArg then sendNetCmd(extraArg)
    elseif action == "COOLKID_EFFECT" then
        task.spawn(function()
            pcall(function()
                local oldSky = Lighting:FindFirstChild("CoolkidSky")
                if oldSky then oldSky:Destroy() end
                local s = Instance.new("Sky", Lighting)
                s.Name = "CoolkidSky"
                for _, f in ipairs({ "SkyboxBk","SkyboxDn","SkyboxFt","SkyboxLf","SkyboxRt","SkyboxUp" }) do
                    s[f] = "http://www.roblox.com/asset/?id=8408806737"
                end
                Lighting.TimeOfDay = 12
            end)
            notify("c00lkidd", "Skybox aplicada!")
        end)
    elseif action == "BACKROOMS_SPAWN" then
        task.spawn(function() pcall(function() spawnBackrooms(extraArg) end) end)
    elseif action == "BACKROOMS_CLEAR" then
        task.spawn(function() pcall(clearBackrooms) end)
    elseif action == "POOLROOMS_SPAWN" then
        task.spawn(function() pcall(function() POOL.spawn(extraArg) end) end)
    elseif action == "POOLROOMS_CLEAR" then
        task.spawn(function() pcall(POOL.clear) end)
    elseif action == "BRAINROT_SPAWN" then
        task.spawn(function() pcall(roubeUmBrainrot, "ON") end)
    elseif action == "BRAINROT_CLEAR" then
        task.spawn(function() pcall(roubeUmBrainrot, "OFF") end)
    elseif action == "LIFEFORM_SPAWN" then
        task.spawn(function()
            local folder = workspace:FindFirstChild("Backrooms_Level0")
            if not folder then
                pcall(function() spawnBackrooms(extraArg) end)
                task.wait(3)
                folder = workspace:FindFirstChild("Backrooms_Level0")
            end
            if folder then
                local targetP = Players:FindFirstChild(extraArg or selectedPlayerName) or LocalPlayer
                local chr = targetP.Character
                local hrp2 = chr and chr:FindFirstChild("HumanoidRootPart")
                local pos = hrp2 and (hrp2.Position + Vector3.new(30, 0, 30)) or Vector3.new(30, 203, 30)
                pcall(function() spawnLifeform(pos, folder, targetP.Name) end)
            end
        end)
    elseif action == "LIFEFORM_CLEAR" then
        task.spawn(function() pcall(clearLifeform) end)
    elseif action == "FLING" then
        task.spawn(function() pcall(doFlingTarget, extraArg or senderName) end)

local recentNet = {}
local function isMe(t)
    if not t then return false end
    local l = string.lower(t)
    return l == string.lower(LocalPlayer.Name) or l == string.lower(LocalPlayer.DisplayName)
end

local function processNetMessage(msg, senderName)
    if not msg then return end
    do
        local now = os.clock()
        local key = tostring(senderName) .. ":" .. msg
        if recentNet[key] and now - recentNet[key] < 1.5 then return end
        recentNet[key] = now
        for k, t in pairs(recentNet) do
            if now - t > 5 then recentNet[k] = nil end
        end
    end
    local body
    if msg:sub(1,1) == ";" then body = msg:sub(2)
    elseif string.lower(msg:sub(1,5)) == "fhub " then body = msg:sub(6)
    else return end
    local parts = string.split(body, " ")
    if #parts < 1 or parts[1] == "" then return end
    local action = string.upper(parts[1])
    local target = parts[2]
    local extraArg = parts[3] and table.concat(parts, " ", 3) or nil

    if action == "PING" then
        local role = (WHITELIST[LocalPlayer.Name] and WHITELIST[LocalPlayer.Name].Role) or "USER"
        sendNetCmd(";PONG " .. LocalPlayer.Name .. " ROLE_" .. role)
        return
    elseif action == "PONG" and target then
        activeScriptUsers[target] = true
        local roleExtra = parts[3] or "ROLE_USER"
        NETWORK_ROLES[target] = roleExtra:gsub("ROLE_", "")
        local pt = Players:FindFirstChild(target)
        if pt and IS_ADMIN then applyOverheadTag(pt) end
        return
    end

    if REQUIRE_WHITELIST and senderName ~= LocalPlayer.Name and not WHITELIST[senderName] then return end

    if target and (string.upper(target) == "ALL" or isMe(target)) then
        task.spawn(function() pcall(executeActionLocally, action, senderName, extraArg) end)
    end
end

local function randf(a, b) return a + (b - a) * math.random() end
local function lerp(a, b, t) return a + (b - a) * t end
local function smooth(t) t = math.clamp(t, 0, 1) return t * t * (3 - 2 * t) end

local function funnelRadius(S, height)
    local n = math.clamp(height / HEIGHT, 0, 1)
    return S.groundRadius + (S.topRadius - S.groundRadius) * (n ^ 1.4)
end

local function funnelOffset(S, height)
    local n = math.clamp(height / HEIGHT, 0, 1)
    local forward = -S.direction * 200 * (n ^ 1.28)
    local sideDir = Vector3.new(-S.direction.Z, 0, S.direction.X)
    local side = sideDir * (math.sin(S.time * 0.35 + n * 4.5) * 50 * n
        + math.sin(S.time * 1.05 + n * 10) * 12 * (1 - n) * S.intensity)
    return forward + side
end

local function axisAt(S, y)
    local h = math.clamp(y - S.groundY, 0, HEIGHT)
    local off = funnelOffset(S, h)
    return Vector3.new(S.center.X + off.X, y, S.center.Z + off.Z), h
end

local function windStrength(S, d, R, mult)
    if not S.touched then return 0 end
    local zone = (R * 2.6 + 170) * (0.5 + 0.5 * S.intensity) * (mult or 1)
    return (math.clamp(1 - d / zone, 0, 1) ^ 0.8) * (0.3 + 0.7 * S.intensity)
end

local function windDirection(rel, d)
    local out = rel / d
    local tangent = Vector3.new(-out.Z, 0, out.X)
    return (tangent * 0.88 - out * 0.48).Unit
end

local function funnelColor(n, core)
    local c
    if n < 0.45 then c = FUNNEL_BOTTOM:Lerp(FUNNEL_MID, n / 0.45)
    else c = FUNNEL_MID:Lerp(FUNNEL_TOP, (n - 0.45) / 0.55) end
    if n < 0.3 then c = c:Lerp(DIRT_MID:Lerp(FUNNEL_BOTTOM, 0.35), ((0.3 - n) / 0.3) * 0.65) end
    if core then c = Color3.new(c.R * 0.5, c.G * 0.5, c.B * 0.52) end
    return c
end

local function updateFunnel(S, dt, refresh)
    for li, layer in ipairs(S.funnel) do
        local height = S.baseHeight + (HEIGHT - S.baseHeight) * layer.nh
        local radius = funnelRadius(S, height) * (1 + 0.08 * math.sin(S.time * 1.5 + li * 0.7))
        local off = funnelOffset(S, height)
        local n = math.clamp(height / HEIGHT, 0, 1)
        layer.angle += (70 / math.max(radius, 20)) * (1 + 0.55 * S.intensity) * dt
        local cx, cz = S.center.X + off.X, S.center.Z + off.Z
        for _, pt in ipairs(layer.points) do
            local rm = pt.core and 0.6 or 1
            local sp = pt.core and 1.38 or 1
            local a = layer.angle * sp + pt.offset
            local r = radius * rm
            local pos = Vector3.new(cx + math.cos(a) * r, S.groundY + height, cz + math.sin(a) * r)
            local tangent = Vector3.new(-math.sin(a), 0, math.cos(a))
            pt.part.CFrame = CFrame.fromMatrix(pos, Vector3.yAxis, tangent)
            if refresh then
                local e = pt.emitter
                e.Enabled = true
                e.Color = ColorSequence.new(funnelColor(n, pt.core))
                local sz = pt.core and math.clamp(radius * 0.6, 18, 140) or math.clamp(radius * 0.9, 26, 200)
                e.Size = NumberSequence.new({
                    NumberSequenceKeypoint.new(0, sz * 0.5),
                    NumberSequenceKeypoint.new(1, sz * 1.4) })
                e.Rate = pt.core and (5 + 9 * S.intensity) or (9 + 16 * S.intensity)
            end
        end
    end
end

local function updateClouds(S, dt, refresh)
    for k, ring in ipairs(S.cloudLayers) do

        ring.angle += (0.3 + 0.05 * k) * dt

        for i, pt in ipairs(ring.points) do
            local a = ring.angle + (i - 1) * math.pi * 2 / #ring.points
            pt.part.Position = Vector3.new(
                S.center.X + off.X + math.cos(a) * radius,
                S.groundY + height,
                S.center.Z + off.Z + math.sin(a) * radius)

                e.Color = ColorSequence.new(k == 1 and EYE_COLOR or CLOUD_COLOR:Lerp(EYE_COLOR, 0.2 * (EYE_RINGS - k) / math.max(EYE_RINGS - 1, 1)))

                    NumberSequenceKeypoint.new(1, size * 1.4) })
                e.Rate = 3 + S.intensity * 4
            end
        end
    end
end

local DECK_FRACS = { 1.25, 1.7, 2.2, 2.8 }
local function updateDeck(S, dt, refresh)
    local off = funnelOffset(S, HEIGHT)
    for k, ring in ipairs(S.deck) do
        ring.angle += (0.12 + 0.03 * k) * dt
        for i, pt in ipairs(ring.points) do
            local a = ring.angle + (i - 1) * math.pi * 2 / #ring.points + pt.jit
            local r = S.topRadius * DECK_FRACS[k] * pt.rm
            pt.part.Position = Vector3.new(
                S.center.X + off.X + math.cos(a) * r,
                S.groundY + pt.alt,
                S.center.Z + off.Z + math.sin(a) * r)
            if refresh then
                local e = pt.emitter
                e.Enabled = true
                e.Color = ColorSequence.new(CLOUD_DARK:Lerp(CLOUD_COLOR, ((k - 1) / (#DECK_FRACS - 1)) * 0.6))
                local size = math.clamp(S.topRadius * 0.4, 140, 420)
                e.Size = NumberSequence.new({
                    NumberSequenceKeypoint.new(0, size * 0.7),
                    NumberSequenceKeypoint.new(1, size * 1.3) })
                e.Rate = (1.2 + 2.2 * S.intensity) * (S.formation or 0)
            end
        end
    end
end

local function updateSandCloud(S, dt, refresh)
    local I = S.intensity
    local gr = math.max(S.groundRadius, 14)
    local active = S.touched
    local dens = active and (0.3 + 0.7 * I) or 0
    for _, d in ipairs(S.dust) do
        local e = d.emitter
        local pos
        if d.kind == "skirt" then
            local r = gr * d.rm + math.sin(S.time * 1.3 + d.phase) * gr * 0.12
            d.angle += d.speed * (60 / math.max(r, 20)) * (0.6 + I) * dt
            pos = Vector3.new(S.center.X + math.cos(d.angle) * r, S.groundY + d.h, S.center.Z + math.sin(d.angle) * r)
            e.Rate = (4 + 12 * I) * dens
            if refresh then
                local s = math.clamp(gr * 0.5, 25, 110)
                e.Size = NumberSequence.new({
                    NumberSequenceKeypoint.new(0, s * 0.6),
                    NumberSequenceKeypoint.new(1, s * 1.5) })
            end
        elseif d.kind == "plume" then
            d.h = (d.h + d.rise * (0.5 + I) * dt) % 280
            local h = d.h
            local r = gr * d.rm * (1 + h / 170)
            d.angle += d.speed * (55 / math.max(r, 20)) * (0.6 + I) * dt
            local off = funnelOffset(S, h)
            pos = Vector3.new(S.center.X + off.X + math.cos(d.angle) * r, S.groundY + h, S.center.Z + off.Z + math.sin(d.angle) * r)
            e.Rate = (3 + 9 * I) * dens * (1 - math.clamp(h / 300, 0, 0.7))
            if refresh then
                local s = math.clamp(gr * 0.35 + h * 0.15, 18, 90)
                e.Size = NumberSequence.new({
                    NumberSequenceKeypoint.new(0, s * 0.5),
                    NumberSequenceKeypoint.new(1, s * 1.4) })
            end
        else
            local r = gr * d.rm
            d.angle += d.speed * 2.2 * (60 / math.max(r, 20)) * (0.7 + I) * dt
            local h = d.h + math.sin(S.time * 3 + d.phase) * 6
            pos = Vector3.new(S.center.X + math.cos(d.angle) * r, S.groundY + math.max(h, 1), S.center.Z + math.sin(d.angle) * r)
            e.Rate = (15 + 55 * I) * dens
            if refresh then
                e.Size = NumberSequence.new({
                    NumberSequenceKeypoint.new(0, 0.7),
                    NumberSequenceKeypoint.new(0.5, 1.4),
                    NumberSequenceKeypoint.new(1, 0.2) })
            end
        end
        d.part.Position = pos
        e.Enabled = active
    end
end

local function releasePlayer(S, hum)
    if S.playerPulled then
        S.playerPulled = false
        pcall(function()
            if hum then
                hum.PlatformStand = false
                hum:ChangeState(Enum.HumanoidStateType.GettingUp)
            end
        end)
    end
end

local function updatePlayer(S, dt)
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum or hum.Health <= 0 then
        S.playerPulled = false
        return
    end
    local pos = hrp.Position
    local axis, h = axisAt(S, pos.Y)
    local rel = Vector3.new(pos.X - axis.X, 0, pos.Z - axis.Z)
    local d = math.max(rel.Magnitude, 0.5)
    local R = funnelRadius(S, h)
    local pullR = R * 1.75 + 50
    local inRange = S.touched and d < (S.playerPulled and pullR * 1.3 or pullR)
    if not inRange then
        releasePlayer(S, hum)
        return
    end
    if not S.playerPulled then
        S.playerPulled = true
        pcall(function() hum:ChangeState(Enum.HumanoidStateType.Physics) end)
    end
    hum.PlatformStand = true
    local I = S.intensity
    local targetVel
    if d < R * 0.35 then
        targetVel = Vector3.new(-rel.X * 1.5, 0, -rel.Z * 1.5) + S.direction * MOVE_SPEED
    else
        local out = rel / d
        local tan = Vector3.new(-out.Z, 0, out.X)
        local radial = math.clamp((R * 0.9 - d) * 0.85, -80, 50)
        targetVel = out * radial + tan * (50 + 95 * I) + Vector3.new(0, h < 500 and (28 + 32 * I) or -10, 0)
    end
    hrp.AssemblyLinearVelocity = hrp.AssemblyLinearVelocity:Lerp(targetVel, 1 - math.exp(-6 * dt))
    hrp.AssemblyAngularVelocity = Vector3.new(math.sin(S.time * 4) * 4, 8 + 12 * I, math.cos(S.time * 3.5) * 4)
end

local function eligiblePart(p)
    if not p:IsA("BasePart") or p:IsA("Terrain") or p:IsA("SpawnLocation") then return false end
    if p.Name == "TornadoPoint" or p.Name == "TornadoSand" then return false end
    local sz = p.Size
    if math.max(sz.X, sz.Y, sz.Z) > 200 then return false end
    if sz.Y < 6 and math.max(sz.X, sz.Z) > 60 then return false end
    if p.Parent and p.Parent:FindFirstChildOfClass("Humanoid") then return false end
    return true
end

local function scanParts(S)
    local I = S.intensity
    if I < 0.05 then return end
    local scanR = (funnelRadius(S, 40) * 2.6 + 170) * (0.5 + 0.5 * I)
    local params = OverlapParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local ex = { S.folder }
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr.Character then table.insert(ex, plr.Character) end
    end
    params.FilterDescendantsInstances = ex
    params.MaxParts = 400
    local parts = workspace:GetPartBoundsInRadius(Vector3.new(S.center.X, S.groundY + 40, S.center.Z), scanR, params)
    local count = 0
    for _ in pairs(S.pulledSet) do count += 1 end
    local added = 0
    for _, p in ipairs(parts) do
        if added >= 25 or count >= MAX_PULLED_PARTS then break end
        if not S.pulledSet[p] and eligiblePart(p) then
            local pos = p.Position
            local axis, h = axisAt(S, pos.Y)
            local rel = Vector3.new(pos.X - axis.X, 0, pos.Z - axis.Z)
            local d = math.max(rel.Magnitude, 0.5)
            local R = funnelRadius(S, h)
            if d < R * 1.8 + 60 then
                pcall(function()
                    for _, j in ipairs(p:GetJoints()) do
                        if j:IsA("JointInstance") or j:IsA("WeldConstraint") then j:Destroy() end
                    end
                    p:BreakJoints()
                end)
                p.Anchored = false
                S.pulledSet[p] = {
                    om = randf(0.6, 1.3), lift = randf(0.8, 1.4),
                    spin = Vector3.new(randf(-5,5), randf(-5,5), randf(-5,5)),
                }
                added += 1
                count += 1
            end
        end
    end
end

local function updateParts(S, dt)
    local I = S.intensity
    if I < 0.05 then return end
    S.scanTimer = (S.scanTimer or 0) + dt
    if S.scanTimer >= 0.35 then
        S.scanTimer = 0
        scanParts(S)
    end
    for p, info in pairs(S.pulledSet) do
        if p and p.Parent then
            local pos = p.Position
            local axis, h = axisAt(S, pos.Y)
            local rel = Vector3.new(pos.X - axis.X, 0, pos.Z - axis.Z)
            local d = math.max(rel.Magnitude, 0.5)
            local R = funnelRadius(S, h)
            local zone = R * 2.6 + 170
            if d > zone * 1.2 then
                pcall(function() p.CanCollide = true end)
                S.pulledSet[p] = nil
            else
                p.CanCollide = false
                local out = rel / d
                local tan = Vector3.new(-out.Z, 0, out.X)
                local radial = math.clamp((R * 0.9 - d) * 0.7, -60, 40)
                local target = out * radial + tan * (35 + 70 * I) * info.om
                    + Vector3.new(0, (15 + 25 * I) * info.lift, 0)
                p.AssemblyLinearVelocity = p.AssemblyLinearVelocity:Lerp(target, 1 - math.exp(-2.5 * dt))
                p.AssemblyAngularVelocity = info.spin * (0.6 + I)
            end
        else
            S.pulledSet[p] = nil
        end
    end
end

local function boltSegment(parent, a, b, thick, color)
    local len = (b - a).Magnitude
    if len < 0.05 then return nil end
    local p = Instance.new("Part")
    p.Anchored, p.CanCollide, p.CanTouch, p.CanQuery, p.CastShadow = true, false, false, false, false
    p.Material, p.Color, p.Transparency = Enum.Material.Neon, color, 1
    p.Size = Vector3.new(thick, thick, len + thick * 0.8)
    p.CFrame = CFrame.lookAt((a + b) * 0.5, b)
    p.Parent = parent
    return p
end

local function boltPath(a, b, segs, jitterMul)
    local pts = { a }
    local dir = b - a
    local len = dir.Magnitude
    local fwd = dir.Unit
    local up = (math.abs(fwd.Y) > 0.9) and Vector3.xAxis or Vector3.yAxis
    local right = fwd:Cross(up).Unit
    local up2 = right:Cross(fwd).Unit
    local step = len / segs
    local jit = step * jitterMul
    local lateral = Vector3.zero
    for i = 1, segs - 1 do
        lateral = lateral * 0.55 + (right * randf(-1, 1) + up2 * randf(-1, 1)) * jit
        pts[#pts + 1] = a + fwd * (step * i) + lateral
    end
    pts[#pts + 1] = b
    return pts
end

local function drawPath(parent, pts, thick, core, glow)
    for i = 1, #pts - 1 do
        local g = boltSegment(parent, pts[i], pts[i+1], thick * 3.2, BOLT_GLOW)
        local c = boltSegment(parent, pts[i], pts[i+1], thick, BOLT_CORE)
        if g then table.insert(glow, g) end
        if c then table.insert(core, c) end
    end
end

local function lightningPoint(S)
    local off = funnelOffset(S, HEIGHT)
    local a = math.random() * math.pi * 2
    local r = S.topRadius * randf(0.9, 2.6)
    local alt = HEIGHT * 0.86 + randf(0, 250)
    return Vector3.new(
        S.center.X + off.X + math.cos(a) * r,
        S.groundY + alt,
        S.center.Z + off.Z + math.sin(a) * r)
end

local function spawnLightning(S)
    local size = math.clamp(S.topRadius * 0.5, 160, 380)
    local r = math.random()
    if r < 0.6 then
        local a = lightningPoint(S)
        local dir = Vector3.new(randf(-1,1), randf(-0.2,0.2), randf(-1,1)).Unit
        local b = a + dir * randf(200, 550)
        local folder = Instance.new("Folder", S.folder)
        local core, glow = {}, {}
        local thick = math.clamp((b-a).Magnitude * 0.0035, 0.5, 4.2)
        drawPath(folder, boltPath(a, b, math.clamp(math.floor((b-a).Magnitude/40), 8, 30), 0.55), thick, core, glow)
        cloudFlash(S, a, size, 1)
        cloudFlash(S, (a + b) * 0.5, size, 0.9)
        thunder(S, (a + b) * 0.5, 1.1)
        task.spawn(function()
            task.wait(0.15)
            for _, p in ipairs(core) do p.Transparency = 1 end
            for _, p in ipairs(glow) do p.Transparency = 1 end
            task.wait(0.2)
            pcall(function() folder:Destroy() end)
        end)
    else
        cloudFlash(S, lightningPoint(S), size * 1.3, 0.9)
        thunder(S, lightningPoint(S), 0.7)
    end
end

-- ============ CLEAR / SPAWN ============
function TORNADO.clear(silent)
    TORNADO.token += 1
    local S = TORNADO.S
    TORNADO.S = nil
    if not S then return end
    if S.connection then pcall(function() S.connection:Disconnect() end) end
    if S.shakeName then pcall(function() RunService:UnbindFromRenderStep(S.shakeName) end) end
    pcall(function()
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        releasePlayer(S, hum)
        if hum then hum.PlatformStand = false end
    end)
    if S.pulledSet then
        for p, _ in pairs(S.pulledSet) do
            pcall(function()
                p.CanCollide = true
                p.AssemblyLinearVelocity = Vector3.zero
                p.AssemblyAngularVelocity = Vector3.zero
            end)
        end
        S.pulledSet = nil
    end
    if S.folder then pcall(function() S.folder:Destroy() end) end
    if S.colorCorrection then pcall(function() S.colorCorrection:Destroy() end) end
    if S.atmosphere then pcall(function() S.atmosphere:Destroy() end) end
    if S.clouds then
        if S.cloudsNew then pcall(function() S.clouds:Destroy() end)
        elseif S.cloudsSaved then
            pcall(function() for k, v in pairs(S.cloudsSaved) do S.clouds[k] = v end end)
        end
    end
    if S.saved then
        for key, value in pairs(S.saved) do pcall(function() Lighting[key] = value end) end
    end
    if not silent then notify("Tornado", "O tornado dissipou.") end
end

function TORNADO.spawn(refName)
    TORNADO.clear(true)
    local refPlayer = (refName and refName ~= LocalPlayer.Name) and Players:FindFirstChild(refName) or nil
    local character = refPlayer and refPlayer.Character
    local hrp = character and character:FindFirstChild("HumanoidRootPart")
    if not hrp then
        character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
        hrp = character:WaitForChild("HumanoidRootPart", 10)
    end
    if not hrp then notify("Tornado", "Personagem nao encontrado.") return end

    local look = hrp.CFrame.LookVector
    look = Vector3.new(look.X, 0, look.Z)
    look = look.Magnitude > 0.1 and look.Unit or Vector3.new(0, 0, -1)
    local startPosition = hrp.Position + look * START_DISTANCE

    local S = {
        time = 0,
        center = Vector3.new(startPosition.X, 0, startPosition.Z),

-- ================== COUCH AUTOMÁTICO v2 ==================
local function getCouch()
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    local character = LocalPlayer.Character
    if not backpack or not character then return nil end

    local function findCouch(container)
        if not container then return nil end
        for _, item in ipairs(container:GetChildren()) do
            if item:IsA("Tool") and (item.Name:lower():find("couch") or item.Name:lower():find("sofa")) then
                return item
            end
        end
        return nil
    end

    local item = findCouch(character) or findCouch(backpack)
    if item then return item end

    local function tryPull()
        pcall(function()
            for _, gui in ipairs(LocalPlayer.PlayerGui:GetDescendants()) do
                if gui:IsA("GuiButton") then
                    local texto = (gui.Name .. " " .. (gui:IsA("TextButton") and gui.Text or "")):lower()
                    if texto:find("couch") or texto:find("sofa") then
                        pcall(function() gui.MouseButton1Click:Fire() end)
                    end
                end
            end
        end)
        pcall(function()
            local re = ReplicatedStorage:FindFirstChild("RE")
            if not re then return end
            local nomes = { "1Tool", "1Too1l", "1Tools", "Tool", "1GiveTool", "GiveTool", "1Tool1" }
            local argSets = {
                { "PickingTools", "Couch" }, { "GiveTool", "Couch" }, { "Couch" },
                { "PickingTools", "Sofa" }, { "GiveTool", "Sofa" },
            }
            for _, rn in ipairs(nomes) do
                local remote = re:FindFirstChild(rn)
                if remote then
                    for _, args in ipairs(argSets) do
                        pcall(function() remote:FireServer(unpack(args)) end)
                        pcall(function() remote:InvokeServer(unpack(args)) end)
                    end
                end
            end
        end)
    end

    local t0 = tick()
    while tick() - t0 < 5 do
        tryPull()
        local tw = tick()
        repeat
            task.wait(0.05)
            item = findCouch(backpack) or findCouch(character)
        until item or tick() - tw > 1
        if item then break end
        task.wait(0.2)
    end

    if item then
        local hum = character:FindFirstChildOfClass("Humanoid")
        if item.Parent == backpack and hum then
            pcall(function() hum:EquipTool(item) end)
        end
    end
    return item
end

local function getPropPart(prop)

local function isBadProp(obj)
    local cur = obj.Parent
    while cur and cur ~= workspace do
        if cur:IsA("Model") and (cur:FindFirstChildOfClass("Humanoid") or cur.Name:find("Car") or cur.Name:find("Bus")) then
            return true
        end
        if cur:IsA("Tool") or cur.Name == "Vehicles" then return true end
        cur = cur.Parent
    end
    return false
end

local mySavedProp = nil

local function getMyProp()
    if mySavedProp and mySavedProp.Parent then
        return mySavedProp
    end

    local hrp = getHRP()
    local best, bestD = nil, math.huge

    for _, obj in ipairs(workspace:GetDescendants()) do
        if (obj:IsA("Model") or obj:IsA("BasePart")) and not isBadProp(obj) then
            local isOwned = obj.Name:find(LocalPlayer.Name) or obj:FindFirstChild("SetCurrentCFrame")
            if isOwned then
                local root = getPropPart(obj)
                if root and hrp then
                    local d = (root.Position - hrp.Position).Magnitude
                    if d < bestD then
                        bestD = d
                        best = obj
                    end
                end
            end
        end
    end

    mySavedProp = best
    return best
end

local function moveProp(prop, cf)
    if not prop or not prop.Parent then return end
    local remote = prop:FindFirstChild("SetCurrentCFrame")

    if remote and remote:IsA("RemoteFunction") then
        pcall(function() remote:InvokeServer(cf) end)
    elseif remote and remote:IsA("RemoteEvent") then
        pcall(function() remote:FireServer(cf) end)
    end

    pcall(function()
        if prop:IsA("Model") then
            if prop.PrimaryPart then prop:SetPrimaryPartCFrame(cf)
            else prop:PivotTo(cf) end
        else
            prop.CFrame = cf
        end
    end)
end

    local re = ReplicatedStorage:FindFirstChild("RE")
    if re and re:FindFirstChild("1Cair") then
        pcall(function() re["1Cair"]:FireServer("PickingCar", "SchoolBus") end)
        task.wait(0.3)
        pcall(function() re["1Cair"]:FireServer("PickingCar", "Bus") end)
    end
    local t0 = tick()
    repeat
        bus = GetBus()
        task.wait(0.3)
    until bus or tick() - t0 > 5
    hrp.CFrame = orig
    return bus
end

-- ================== BUS COM MOTORISTA ==================
local function doBusDriverKill(targetPlayer)
    if not targetPlayer or targetPlayer == LocalPlayer then return end
    applyNetworkBypass()
    local tc = targetPlayer.Character
    local tr = tc and tc:FindFirstChild("HumanoidRootPart")
    local thum = tc and tc:FindFirstChildOfClass("Humanoid")
    if not tr or not thum then return end
    local myChr = LocalPlayer.Character
    local myHrp = getHRP()
    local myHum = myChr and myChr:FindFirstChildOfClass("Humanoid")
    if not myHrp or not myHum then return end
    local originalCFrame = myHrp.CFrame

    local bus = SpawnBus()
    if not bus then return end
    local driverSeat = bus:FindFirstChildWhichIsA("VehicleSeat", true)
    if not driverSeat then driverSeat = bus:FindFirstChildWhichIsA("Seat", true) or bus.PrimaryPart end
    if not driverSeat then return end

    myHrp.CFrame = driverSeat.CFrame * CFrame.new(0, 2, 0)
    task.wait(0.2)
    pcall(function() driverSeat:Sit(myHum) end)
    task.wait(0.4)

    local angle, startTime = 0, tick()
    while tick() - startTime < 4.5 do
        if not tr or not tr.Parent or thum.Health <= 0 then break end
        applyNetworkBypass()
        angle = angle + 45
        pcall(function()
            bus:PivotTo(tr.CFrame * CFrame.Angles(0, math.rad(angle), 0))
            driverSeat.AssemblyLinearVelocity = Vector3.new(0, -10, 0)
        end)
        if thum.Sit then break end
        task.wait()
    end

    local dropStart = tick()
    while tick() - dropStart < 3.5 do
        if not tr or not tr.Parent or thum.Health <= 0 then break end
        applyNetworkBypass()
        pcall(function()
            driverSeat.AssemblyLinearVelocity = Vector3.new(0, -FALL_SPEED, 0)
            tr.AssemblyLinearVelocity = Vector3.new(0, -FALL_SPEED, 0)
            bus:PivotTo(CFrame.new(tr.Position))
        end)
        task.wait()
        if tr.Position.Y <= VOID_Y then break end
    end

    pcall(function() myHum.Sit = false end)
    task.wait(0.1)
    if myHrp and myHrp.Parent then
        myHrp.CFrame = originalCFrame
        myHrp.AssemblyLinearVelocity = Vector3.zero
        myHrp.AssemblyAngularVelocity = Vector3.zero
    end
    pcall(function() thum.Health = 0; tr:BreakJoints() end)
    pcall(function()
        if ReplicatedStorage.RE and ReplicatedStorage.RE:FindFirstChild("1Cair") then
            ReplicatedStorage.RE["1Cair"]:FireServer("DeleteAllVehicles")
        end
    end)
end

-- ================== FLING BALL ==================
local function executeFlingBall(targetPlayer)
    if not targetPlayer or targetPlayer == LocalPlayer then return end
    applyNetworkBypass()
    local tc = targetPlayer.Character
    local tr = tc and tc:FindFirstChild("HumanoidRootPart")
    local thum = tc and tc:FindFirstChildOfClass("Humanoid")
    if not tr or not thum then return end

    local ballTool = getSoccerBall()
    if not ballTool then return end
    local character = LocalPlayer.Character
    local myHum = character and character:FindFirstChildOfClass("Humanoid")
    if myHum then pcall(function() myHum:EquipTool(ballTool) end) end
    task.wait(0.2)

    local handle = ballTool:FindFirstChild("Handle") or ballTool:FindFirstChildWhichIsA("BasePart")
    if not handle then return end
    ballTool.Parent = workspace
    handle.CanCollide = false

    local bv = handle:FindFirstChildOfClass("BodyVelocity") or Instance.new("BodyVelocity")
    bv.Name = "FlingVelocity"
    bv.Velocity = Vector3.new(9e8, 9e8, 9e8)
    bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bv.P = 9e9
    bv.Parent = handle

    local startTime = tick()
    while tick() - startTime < 3 and tr and tr.Parent and thum.Health > 0 do
        applyNetworkBypass()
        pcall(function()
            handle.CFrame = tr.CFrame
            handle.AssemblyLinearVelocity = Vector3.new(math.random(-9e8, 9e8), 9e8, math.random(-9e8, 9e8))
            tr.AssemblyLinearVelocity = Vector3.new(0, 9e8, 0)
        end)
        task.wait()
    end
    pcall(function() ballTool:Destroy() end)
end

-- ================== PROPMAKER KILL (100% SEGURO SEM TELEPORTE DO SEU PLAYER) ==================
local function executePropMakerKill(targetPlayer)
    if not targetPlayer or targetPlayer == LocalPlayer then return end
    applyNetworkBypass()

    local tc = targetPlayer.Character
    local tr = tc and tc:FindFirstChild("HumanoidRootPart")
    local thum = tc and tc:FindFirstChildOfClass("Humanoid")
    if not tr or not thum or thum.Health <= 0 then return end

    local myHrp = getHRP()
    if not myHrp then return end
    local safePos = myHrp.CFrame

    local prop = getMyProp()
    if not prop then return end

    local propRoot = getPropPart(prop)
    if not propRoot then return end

    -- TRAVA O PERSONAGEM DO PLAYER LOCAL NA POSIÇÃO SEGURA
    local holdConnection
    holdConnection = RunService.Heartbeat:Connect(function()
        if myHrp and myHrp.Parent then
            myHrp.CFrame = safePos
            myHrp.AssemblyLinearVelocity = Vector3.zero
            myHrp.AssemblyAngularVelocity = Vector3.zero
        end
    end)

    -- Puxa e empurra o prop para o alvo sem mover seu jogador
    local dropStart = tick()
    local curTargetCF = tr.CFrame

    while tick() - dropStart < 3 do
        if not tr or not tr.Parent or thum.Health <= 0 then break end
        applyNetworkBypass()

        curTargetCF = curTargetCF * CFrame.new(0, -85, 0)

        pcall(function()
            -- Aplica forca física na peça do prop
            if propRoot and propRoot.Parent then
                propRoot.AssemblyLinearVelocity = Vector3.new(0, -FALL_SPEED, 0)
                propRoot.AssemblyAngularVelocity = Vector3.new(500, 500, 500)
            end
            
            -- Puxa o alvo para baixo
            tr.AssemblyLinearVelocity = Vector3.new(0, -FALL_SPEED, 0)
            
            -- Envia a ordem de movimento do prop via Remote para a posição do alvo
            moveProp(prop, curTargetCF)
        end)

        task.wait(0.02)
        if curTargetCF.Y <= VOID_Y then break end
    end

    -- Desconecta o travamento do seu jogador
    if holdConnection then holdConnection:Disconnect() end

    -- Mata o alvo
    pcall(function() thum.Health = 0; tr:BreakJoints() end)

    -- Restaura a velocidade do seu personagem
    if myHrp and myHrp.Parent then
        myHrp.CFrame = safePos
        myHrp.AssemblyLinearVelocity = Vector3.zero
        myHrp.AssemblyAngularVelocity = Vector3.zero
    end

    -- Retorna o prop para perto de você
    if propRoot and propRoot.Parent then
        pcall(function()
            moveProp(prop, safePos * CFrame.new(0, 4, 0))
            propRoot.AssemblyLinearVelocity = Vector3.zero
            propRoot.AssemblyAngularVelocity = Vector3.zero
        end)
    end
end
